2// nanom/values.hpp — value kernels shared by columnar readers (Parquet, Lance, Arrow IPC …).
3//
4// columnar.hpp turns encoded bits into integers (bit unpacking, RLE / bit-packed runs, deltas).
5// This header is the step after: turning those integers and byte ranges into the values a reader
6// hands out — validity bitmaps, fixed-width values spread over null slots, variable-length values
7// in offsets + data layout (Arrow's, and Lance's), dictionary gathers, UTF-8 validation and
8// integer / decimal widening. Nothing here knows a file format.
9//
10// Same safety rule as columnar.hpp — CHECK AT THE BOUNDARY, RUN UNCHECKED INSIDE: every length,
11// index and count that came from the wire is validated before it is used, and a kernel that
12// cannot finish says why (a static message) instead of producing a partial guess.
13//
14// bitmaps set_bits, count_bits, get_bit (LSB-first, Arrow's validity layout),
15// rle_bitmap (a width-1 RLE / bit-packed stream -> bitmap, run by run),
16// scatter_bits (dense bits -> the non-null slots of a bitmap)
17// levels decode_levels (RLE / bit-packed levels -> u16, run by run, range-checked),
18// level_slots (Dremel definition levels -> value slots + their validity),
19// struct_slots / list_slots (Dremel record assembly: validity, list offsets)
20// utf8 valid_utf8 (ASCII-word fast path), utf8_starts_ok (values start on a boundary)
21// nulls spread_nulls: nn dense fixed-width values -> n slots, in place, nulls zeroed
22// widening sign_extend_be (big-endian two's complement -> little-endian, any width),
23// sign_extend_le (signed integer -> wider little-endian two's complement)
24// byte arrays length_prefixed_views (u32 LE length + bytes, i.e. Parquet PLAIN BYTE_ARRAY),
25// delta_length_views, delta_prefix_views (DELTA_LENGTH_BYTE_ARRAY / DELTA_BYTE_ARRAY)
26// dictionaries check_indices, gather_fixed, gathered_size
27// offsets append_length_prefixed, append_gathered, append_views: write values + int32
28// offsets for n slots (null slots repeat the offset) into an offsets + data pair

51inline bool get_bit(const std::uint8_t* bm, std::size_t i) { return (bm[i / 8] >> (i % 8)) & 1; }
52
54inline void set_bits(std::uint8_t* bm, std::size_t start, std::size_t count) {
55 std::size_t i = start;
56 const std::size_t end = start + count;
57 for (; i < end && i % 8; ++i) bm[i / 8] = std::uint8_t(bm[i / 8] | (1u << (i % 8)));
58 if (end - i >= 8) {
59 std::memset(bm + i / 8, 0xff, (end - i) / 8);
60 i += ((end - i) / 8) * 8;
61 }
62 for (; i < end; ++i) bm[i / 8] = std::uint8_t(bm[i / 8] | (1u << (i % 8)));
63}
64
66inline std::size_t count_bits(const std::uint8_t* bm, std::size_t start, std::size_t count) {
67 std::size_t c = 0, i = start;
68 const std::size_t end = start + count;
69 for (; i < end && i % 64; ++i) c += get_bit(bm, i);
70 for (; end - i >= 64; i += 64) {
71 std::uint64_t w;
72 std::memcpy(&w, bm + i / 8, 8);
73 c += std::size_t(std::popcount(w));
74 }
75 for (; i < end; ++i) c += get_bit(bm, i);
76 return c;
77}

83inline kernel_status rle_bitmap(rle_bp_decoder& d, std::size_t n, std::span<std::byte> out, std::size_t at) {
84 if (d.width() != 1) return kernel_fail("rle_bitmap needs a width-1 stream");
85 if (at > out.size() * 8 || n > out.size() * 8 - at) return kernel_fail("bitmap smaller than the value count");
86 auto* bm = reinterpret_cast<std::uint8_t*>(out.data());
87 rle_bp_decoder::run lr;
88 for (std::size_t i = 0; i < n; i += lr.count) {
89 if (!d.next_run(lr, n - i)) return kernel_fail("bit stream shorter than its values");
90 if (!lr.packed) {
91 if (lr.value > 1) return kernel_fail("value above 1 in a bit stream");
92 if (lr.value) set_bits(bm, at + i, lr.count);
93 } else if (lr.bit_offset == 0) {
94 if (!copy_bits(lr.bits, lr.count, out, at + i)) return kernel_fail("bit stream shorter than its values");
95 } else { // a run resumed mid-byte
96 for (std::size_t k = 0; k < lr.count; ++k) {
97 const std::size_t b = lr.bit_offset + k;
98 if ((std::uint8_t(lr.bits[b / 8]) >> (b % 8)) & 1) set_bits(bm, at + i + k, 1);
99 }
100 }
101 }
102 return kernel_ok;
103}

108inline void scatter_bits(const std::uint8_t* dense, const std::uint8_t* validity, std::size_t n, std::uint8_t* out,
109 std::size_t at) {
110 std::size_t j = 0;
111 for (std::size_t r = 0; r < n; ++r) {
112 if (validity && !get_bit(validity, r)) continue;
113 const std::size_t b = at + r;
114 out[b / 8] = std::uint8_t(out[b / 8] | (unsigned(get_bit(dense, j)) << (b % 8)));
115 ++j;
116 }
117}

125inline kernel_status decode_levels(std::span<const std::byte> data, std::uint16_t maxv, std::size_t n, std::uint16_t* out) {
126 const unsigned w = unsigned(std::bit_width(unsigned(maxv)));
127 rle_bp_decoder d(data, w);
128 rle_bp_decoder::run lr;
129 std::uint32_t tmp[512];
130 for (std::size_t i = 0; i < n; i += lr.count) {
131 if (!d.next_run(lr, n - i)) return kernel_fail("levels shorter than the page");
132 if (!lr.packed) {
133 if (lr.value > maxv) return kernel_fail("level above the column's maximum");
134 std::fill_n(out + i, lr.count, std::uint16_t(lr.value));
135 continue;
136 }
137 std::uint32_t hi = 0;
138 if (lr.bit_offset == 0) {
139 for (std::size_t k = 0; k < lr.count; k += 512) {
140 const std::size_t m = std::min<std::size_t>(512, lr.count - k);
141 const std::size_t byte = k * w / 8; // k is a multiple of 8: a whole-byte position
142 if (!unpack_bits<std::uint32_t>(lr.bits.subspan(byte), w, std::span<std::uint32_t>(tmp), m))
143 return kernel_fail("levels shorter than the page");
144 for (std::size_t j = 0; j < m; ++j) {
145 hi |= tmp[j] > maxv ? 1u : 0u;
146 out[i + k + j] = std::uint16_t(tmp[j]);
147 }
148 }
149 } else { // a run resumed mid-byte (not produced by one pass, kept for generality)
150 const auto* b = reinterpret_cast<const std::uint8_t*>(lr.bits.data());
151 for (std::size_t k = 0; k < lr.count; ++k) {
152 const std::size_t bit = lr.bit_offset + k * w;
153 std::uint32_t v = 0;
154 for (unsigned t = 0; t < w; ++t) {
155 const std::size_t q = bit + t;
156 if (q / 8 >= lr.bits.size()) return kernel_fail("levels shorter than the page");
157 v |= std::uint32_t((b[q / 8] >> (q % 8)) & 1) << t;
158 }
159 hi |= v > maxv ? 1u : 0u;
160 out[i + k] = std::uint16_t(v);
161 }
162 }
163 if (hi) return kernel_fail("level above the column's maximum");
164 }
165 return kernel_ok;
166}

172struct slot_counts {
173 std::size_t slots = 0, non_null = 0;
174};
175inline slot_counts level_slots(const std::uint16_t* def, std::size_t n, std::uint16_t slot_level, std::uint16_t max_def,
176 std::uint8_t* bm) {
177 slot_counts c;
178 for (std::size_t i = 0; i < n; ++i) {
179 if (def[i] < slot_level) continue; // an empty or null ancestor: no slot here
180 if (def[i] == max_def) {
181 bm[c.slots / 8] = std::uint8_t(bm[c.slots / 8] | (1u << (c.slots % 8)));
182 ++c.non_null;
183 }
184 ++c.slots;
185 }
186 return c;
187}

189// ---------------------------------------------------------------------------
190// 34c. Dremel record assembly (structure from one leaf's levels)
191// ---------------------------------------------------------------------------
192//
193// For a node whose enclosing list has repetition level r_enc and element level d_enc (both 0 at
194// the top level), the node's slots are the level entries with rep <= r_enc and def >= d_enc:
195// struct slot non-null when def >= def_present
196// list slot non-null when def >= def_present; holds an element when def >= def_elem, and
197// every later entry with rep == rep_elem (before the next slot) is another element
198// The caller knows how many slots the parent promised (`expected`): more or fewer is an error,
199// so leaves that disagree about the structure are rejected rather than assembled inconsistently.
200
201struct dremel_node {
202 std::uint16_t r_enc = 0, d_enc = 0;
203 std::uint16_t def_present = 0;
204 std::uint16_t def_elem = 0, rep_elem = 0;
205};
206
209inline kernel_status struct_slots(const std::uint16_t* rep, const std::uint16_t* def, std::size_t entries,
210 const dremel_node& nd, std::int64_t expected, std::uint8_t* validity,
211 std::int64_t& nulls) {
212 std::int64_t slots = 0;
213 nulls = 0;
214 for (std::size_t i = 0; i < entries; ++i) {
215 if (rep[i] > nd.r_enc || def[i] < nd.d_enc) continue;
216 if (slots >= expected) return kernel_fail("more slots than the parent has");
217 if (def[i] >= nd.def_present) validity[slots / 8] = std::uint8_t(validity[slots / 8] | (1u << (slots % 8)));
218 else ++nulls;
219 ++slots;
220 }
221 if (slots != expected) return kernel_fail("fewer slots than the parent has");
222 return kernel_ok;
223}

228inline kernel_status list_slots(const std::uint16_t* rep, const std::uint16_t* def, std::size_t entries,
229 const dremel_node& nd, std::int64_t expected, std::int32_t* offsets,
230 std::uint8_t* validity, std::int64_t& nulls, std::int64_t& elements) {
231 std::int64_t slots = 0, elems = 0;
232 nulls = 0;
233 bool open = false; // the current slot holds a non-empty list
234 for (std::size_t i = 0; i < entries; ++i) {
235 if (rep[i] <= nd.r_enc) {
236 open = false;
237 if (def[i] < nd.d_enc) continue; // an empty / null ancestor: not a slot of this list
238 if (slots >= expected) return kernel_fail("more slots than the parent has");
239 offsets[slots] = std::int32_t(elems);
240 if (def[i] >= nd.def_present) validity[slots / 8] = std::uint8_t(validity[slots / 8] | (1u << (slots % 8)));
241 else ++nulls;
242 ++slots;
243 if (def[i] >= nd.def_elem) {
244 open = true;
245 ++elems;
246 }
247 } else if (rep[i] == nd.rep_elem) {
248 if (!open) return kernel_fail("an element continues a list that has none");
249 ++elems;
250 }
251 if (elems > INT32_MAX) return kernel_fail("more than 2^31 list elements");
252 }
253 if (slots != expected) return kernel_fail("fewer slots than the parent has");
254 offsets[slots] = std::int32_t(elems);
255 elements = elems;
256 return kernel_ok;
257}

265inline bool valid_utf8(std::string_view s) {
266 const auto* p = reinterpret_cast<const unsigned char*>(s.data());
267 const auto* e = p + s.size();
268 while (p < e) {
269 if (e - p >= 16) {
270 std::uint64_t w0, w1;
271 std::memcpy(&w0, p, 8);
272 std::memcpy(&w1, p + 8, 8);
273 if (!((w0 | w1) & 0x8080808080808080ull)) { p += 16; continue; }
274 } else if (e - p >= 8) {
275 std::uint64_t w;
276 std::memcpy(&w, p, 8);
277 if (!(w & 0x8080808080808080ull)) { p += 8; continue; }
278 }
279 const unsigned c = *p;
280 if (c < 0x80) { ++p; continue; }
281 int n;
282 std::uint32_t cp;
283 if ((c & 0xe0) == 0xc0) { n = 1; cp = c & 0x1f; }
284 else if ((c & 0xf0) == 0xe0) { n = 2; cp = c & 0x0f; }
285 else if ((c & 0xf8) == 0xf0) { n = 3; cp = c & 0x07; }
286 else return false;
287 if (e - p <= n) return false;
288 for (int i = 1; i <= n; ++i) {
289 if ((p[i] & 0xc0) != 0x80) return false;
290 cp = (cp << 6) | (p[i] & 0x3f);
291 }
292 if ((n == 1 && cp < 0x80) || (n == 2 && cp < 0x800) || (n == 3 && cp < 0x10000) || cp > 0x10ffff ||
293 (cp >= 0xd800 && cp <= 0xdfff))
294 return false;
295 p += n + 1;
296 }
297 return true;
298}

303template <class Off>
304inline bool utf8_starts_ok(const std::byte* data, std::int64_t end, const Off* starts, std::size_t n) {
305 for (std::size_t i = 0; i < n; ++i) {
306 const auto o = std::int64_t(starts[i]);
307 if (o < end && (std::uint8_t(data[o]) & 0xc0) == 0x80) return false;
308 }

320template <std::size_t W>
321inline void spread_nulls_w(std::byte* dst, std::size_t n, std::size_t nn, const std::uint8_t* bm) {
322 std::size_t j = nn, r = n;
323 while (r > 0) {
324 if (j == 0) { // only nulls remain
325 std::memset(dst, 0, r * W);
326 return;
327 }
328 if (r % 64 == 0) {
329 std::uint64_t word;
330 std::memcpy(&word, bm + (r - 64) / 8, 8);
331 if (word == ~std::uint64_t(0)) {
332 j -= 64;
333 r -= 64;
334 if (j != r) std::memmove(dst + r * W, dst + j * W, 64 * W);
335 continue;
336 }
337 if (word == 0) {
338 r -= 64;
339 std::memset(dst + r * W, 0, 64 * W);
340 continue;
341 }
342 if (j >= 64) { // mixed word: j >= popcount(word) >= 1 keeps dst[j - 1] in range
343 for (int k = 0; k < 64; ++k) {
344 --r;
345 const std::size_t bit = (word >> 63) & 1;
346 word <<= 1;
347 std::byte v[W];
348 std::memcpy(v, dst + (j - 1) * W, W);
349 if constexpr (W <= 8) {
350 using U = std::conditional_t<W == 1, std::uint8_t, std::conditional_t<W == 2, std::uint16_t,
351 std::conditional_t<W <= 4, std::uint32_t, std::uint64_t>>>;
352 U u = 0;
353 std::memcpy(&u, v, W);
354 u = U(u & (U(0) - U(bit)));
355 std::memcpy(dst + r * W, &u, W);
356 } else {
357 const std::byte m = bit ? std::byte{0xff} : std::byte{0};
358 for (std::size_t b = 0; b < W; ++b) v[b] &= m;
359 std::memcpy(dst + r * W, v, W);
360 }
361 j -= bit;
362 }
363 continue;
364 }
365 }
366 --r;
367 if (get_bit(bm, r)) {
368 --j;
369 if (j != r) std::memcpy(dst + r * W, dst + j * W, W);
370 } else {
371 std::memset(dst + r * W, 0, W);
372 }
373 }
374}
375} // namespace detail

380inline void spread_nulls(std::byte* dst, std::size_t n, std::size_t nn, std::size_t width,
381 const std::uint8_t* validity) {
382 switch (width) {
383 case 1: return detail::spread_nulls_w<1>(dst, n, nn, validity);
384 case 2: return detail::spread_nulls_w<2>(dst, n, nn, validity);
385 case 4: return detail::spread_nulls_w<4>(dst, n, nn, validity);
386 case 8: return detail::spread_nulls_w<8>(dst, n, nn, validity);
387 case 16: return detail::spread_nulls_w<16>(dst, n, nn, validity);
388 default: break;
389 }
390 std::size_t j = nn;
391 for (std::size_t r = n; r-- > 0;) {
392 if (get_bit(validity, r)) {
393 --j;
394 if (j != r) std::memcpy(dst + r * width, dst + j * width, width);
395 } else {
396 std::memset(dst + r * width, 0, width);
397 }
398 }
399}

407inline void sign_extend_be(const std::byte* src, std::size_t k, std::byte* out, std::size_t width) {
408 const std::byte sign = k && (std::uint8_t(src[0]) & 0x80) ? std::byte{0xff} : std::byte{0};
409 for (std::size_t i = 0; i < width; ++i) out[i] = i < k ? src[k - 1 - i] : sign;
410}

413template <std::signed_integral T>
414inline void sign_extend_le(T v, std::byte* out, std::size_t width) {
415 std::memcpy(out, &v, sizeof(T));
416 std::memset(out + sizeof(T), v < 0 ? 0xff : 0, width - sizeof(T));
417}

425inline std::optional<std::size_t> length_prefixed_views(std::span<const std::byte> in, std::size_t nn,
426 std::span<std::string_view> out) {
427 if (out.size() < nn) return std::nullopt;
428 std::size_t p = 0;
429 for (std::size_t i = 0; i < nn; ++i) {
430 if (in.size() - p < 4) return std::nullopt;
431 std::uint32_t len;
432 std::memcpy(&len, in.data() + p, 4);
433 if constexpr (std::endian::native == std::endian::big) len = std::byteswap(len);
434 p += 4;
435 if (len > in.size() - p) return std::nullopt;
436 out[i] = std::string_view(reinterpret_cast<const char*>(in.data()) + p, len);
437 p += len;
438 }
439 return p;
440}

444inline kernel_status delta_length_views(std::span<const std::byte> in, std::size_t nn,
445 std::vector<std::int32_t>& lengths, std::span<std::string_view> out) {
446 if (out.size() < nn) return kernel_fail("output smaller than the value count");
447 lengths.resize(nn);
448 auto used = delta_binary_packed<std::int32_t>(in, std::span<std::int32_t>(lengths), nn);
449 if (!used) return kernel_fail("malformed DELTA_LENGTH_BYTE_ARRAY lengths");
450 std::size_t p = *used;
451 for (std::size_t i = 0; i < nn; ++i) {
452 if (lengths[i] < 0 || std::size_t(lengths[i]) > in.size() - p)
453 return kernel_fail("DELTA_LENGTH_BYTE_ARRAY data past the page");
454 out[i] = std::string_view(reinterpret_cast<const char*>(in.data()) + p, std::size_t(lengths[i]));
455 p += std::size_t(lengths[i]);
456 }
457 return kernel_ok;
458}

464inline kernel_status delta_prefix_views(std::span<const std::byte> in, std::size_t nn, std::vector<std::int32_t>& prefix,
465 std::vector<std::int32_t>& lengths, std::vector<char>& arena,
466 std::span<std::string_view> out, std::size_t max_bytes) {
467 if (out.size() < nn) return kernel_fail("output smaller than the value count");
468 prefix.resize(nn);
469 lengths.resize(nn);
470 auto u1 = delta_binary_packed<std::int32_t>(in, std::span<std::int32_t>(prefix), nn);
471 if (!u1) return kernel_fail("malformed DELTA_BYTE_ARRAY prefix lengths");
472 const auto rest = in.subspan(*u1);
473 auto u2 = delta_binary_packed<std::int32_t>(rest, std::span<std::int32_t>(lengths), nn);
474 if (!u2) return kernel_fail("malformed DELTA_BYTE_ARRAY suffix lengths");
475 // size the arena once, validating every prefix / suffix on the way
476 std::size_t total = 0, prev = 0, p = *u2;
477 for (std::size_t i = 0; i < nn; ++i) {
478 if (prefix[i] < 0 || lengths[i] < 0 || std::size_t(prefix[i]) > prev || std::size_t(lengths[i]) > rest.size() - p)
479 return kernel_fail("DELTA_BYTE_ARRAY prefix/suffix out of range");
480 p += std::size_t(lengths[i]);
481 prev = std::size_t(prefix[i]) + std::size_t(lengths[i]);
482 if (prev > max_bytes || total > max_bytes - prev) return kernel_fail("DELTA_BYTE_ARRAY expands past the size limit");
483 total += prev;
484 }
485 arena.resize(total);
486 std::size_t at = 0, prev_at = 0;
487 p = *u2;
488 for (std::size_t i = 0; i < nn; ++i) {
489 const std::size_t pre = std::size_t(prefix[i]), suf = std::size_t(lengths[i]);
490 if (pre) std::memmove(arena.data() + at, arena.data() + prev_at, pre);
491 if (suf) std::memcpy(arena.data() + at + pre, rest.data() + p, suf);
492 p += suf;
493 out[i] = std::string_view(arena.data() + at, pre + suf);
494 prev_at = at;
495 at += pre + suf;
496 }
497 return kernel_ok;
498}

506 std::uint32_t bad = 0;

512inline void gather_fixed(const std::byte* dict, std::size_t width, std::span<const std::uint32_t> idx, std::byte* out) {
513 const std::size_t k = idx.size();
514 if (width == 4) for (std::size_t i = 0; i < k; ++i) std::memcpy(out + 4 * i, dict + 4 * std::size_t(idx[i]), 4);
515 else if (width == 8) for (std::size_t i = 0; i < k; ++i) std::memcpy(out + 8 * i, dict + 8 * std::size_t(idx[i]), 8);
516 else for (std::size_t i = 0; i < k; ++i) std::memcpy(out + width * i, dict + width * std::size_t(idx[i]), width);
517}

524 for (const auto i : idx) total += std::uint64_t(dict_offsets[i + 1] - dict_offsets[i]);

528// ---------------------------------------------------------------------------
529// 40. offsets + data output (Arrow's / Lance's variable-length layout)
530// ---------------------------------------------------------------------------
531//
532// The append_* kernels fill n slots of an offsets + data pair: slot r's END offset goes to
533// ends[r] (so pass offsets + first_slot + 1), valid slots take the next value, null slots
534// (validity bit clear; null validity = all valid) repeat the offset. `cur` is the data size
535// before and after. Each takes kValueSlack writable bytes past the data it writes: short values
536// move as one fixed 16-byte copy.
537
538inline constexpr std::size_t kValueSlack = 16;
539
540namespace detail {
543inline void copy_value(std::byte* dst, const std::byte* src, std::size_t len, const std::byte* src_end) {
544 if (len <= 16 && src_end - src >= 16) std::memcpy(dst, src, 16);
545 else if (len) std::memcpy(dst, src, len);
546}
547inline bool slot_valid(const std::uint8_t* validity, std::size_t r) { return !validity || get_bit(validity, r); }
548} // namespace detail

552template <class Off>
553inline kernel_status append_length_prefixed(std::span<const std::byte> in, std::size_t n, const std::uint8_t* validity,
554 std::byte* data, std::int64_t& cur, Off* ends, std::int64_t max_size) {
555 const std::byte* p = in.data();
556 const std::byte* const e = p + in.size();
557 for (std::size_t r = 0; r < n; ++r) {
558 if (detail::slot_valid(validity, r)) {
559 if (e - p < 4) return kernel_fail("truncated length-prefixed value");
560 std::uint32_t len;
561 std::memcpy(&len, p, 4);
562 if constexpr (std::endian::native == std::endian::big) len = std::byteswap(len);
563 p += 4;
564 if (len > std::size_t(e - p)) return kernel_fail("length-prefixed value runs past the page");
565 detail::copy_value(data + cur, p, len, e);
566 p += len;
567 cur += len;
568 if (cur > max_size) return kernel_fail("values exceed the offset type");
569 }
570 ends[r] = Off(cur);
571 }
572 return kernel_ok;
573}

578template <class Off, class DOff>
579inline void append_gathered(const std::byte* dict_data, const std::byte* dict_end, const DOff* dict_offsets,
580 std::span<const std::uint32_t> idx, std::size_t n, const std::uint8_t* validity,
581 std::byte* data, std::int64_t& cur, Off* ends) {
582 std::size_t j = 0;
583 for (std::size_t r = 0; r < n; ++r) {
584 if (detail::slot_valid(validity, r)) {
585 const auto a = std::size_t(dict_offsets[idx[j]]);
586 const auto len = std::size_t(dict_offsets[idx[j] + 1]) - a;
587 ++j;
588 detail::copy_value(data + cur, dict_data + a, len, dict_end);
589 cur += std::int64_t(len);
590 }
591 ends[r] = Off(cur);
592 }
593}

596template <class Off>
597inline void append_views(std::span<const std::string_view> views, std::size_t n, const std::uint8_t* validity,
598 std::byte* data, std::int64_t& cur, Off* ends) {
599 std::size_t j = 0;
600 for (std::size_t r = 0; r < n; ++r) {
601 if (detail::slot_valid(validity, r)) {
602 const auto v = views[j++];
603 if (!v.empty()) std::memcpy(data + cur, v.data(), v.size());
604 cur += std::int64_t(v.size());
605 }
606 ends[r] = Off(cur);
607 }
