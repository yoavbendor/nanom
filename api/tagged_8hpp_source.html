2// nanom/tagged.hpp — reflected TAGGED messages: self-describing, field-numbered wire formats
3// (Thrift compact today; protobuf rides the same model next). An "extra", layered on reflect.hpp.
4//
5// reflect.hpp describes FIXED-LAYOUT structs: every field at a compile-time bit offset. Columnar file
6// formats keep their metadata in TAGGED encodings instead — Parquet's footer and page headers are
7// Thrift compact, Lance's manifests and page layouts are protobuf: each value is preceded by a field
8// id + wire type, fields may be absent, repeated or unknown, and strings/lists are length-prefixed.
9// This header gives those formats the same "the struct IS the schema" treatment:
10//
11// struct KeyValue {
12// nm::field<1, std::string_view> key; // required (absent = error)
13// nm::field<2, std::optional<std::string_view>> value; // optional (absent = nullopt)
14// };
15// NANOM_DESCRIBE(KeyValue, key, value); // C++23; under C++26 reflection no registration at all
16//
17// auto r = nm::thrift_compact<KeyValue>()(in); // an ordinary nanom parser
18//
19// This header only READS. Writing the same model back (nm::thrift_compact_encode) lives in
20// nanom/tagged_encode.hpp, which a reader never needs to include.
21//
22// The field id lives in the TYPE (field<Id, T>), exactly as endianness lives in be<>/le<> — so both
23// describe<T> providers (the NANOM_DESCRIBE macro and C++26 P2996 reflection) work unchanged, and
24// everything the codec needs is known at compile time:
25//
26// * a field-id -> member dispatch table (dense array for ids < 256), built by consteval code;
27// * the expected wire type of every member: a field whose wire type does not match is never
28// reinterpreted — it is skipped like an unknown field (Apache Thrift's own semantics, which
29// real files rely on: old writers reused field ids with other types), and if the member is
30// required the message then fails the required-field check;
31// * the required-field bitmask, checked once when the struct's STOP byte arrives;
32// * id uniqueness / ascending order / ≤ 64 fields — static_asserts, not runtime surprises.
33//
34// Zero-copy: std::string_view and nm::bytes members point into the input buffer (same lifetime
35// contract as every other nanom view; prefer nm::bytes under NANOM_GENERATION — it carries the
36// attestation, a string_view cannot). nm::list<E> is a LAZY list (the validated element bytes +
37// count; decode on iteration, no allocation) and nm::lazy<M> a lazily decoded nested struct — reading
38// one row group out of a 10,000-row-group Parquet footer touches nothing else. std::vector<E> and
39// std::string are accepted too when an owning copy is what you want.
40//
41// Hostile-input posture (all bounded, nothing allocates from a wire count before checking it):
42// * every list count is checked against the remaining bytes (count_fits) before any reserve;
43// * every length prefix is checked against the remaining bytes before any span is formed;
44// * nesting depth (struct-in-list-in-struct…) is capped at max_tagged_depth, for decode and skip;
45// * integers are range-checked into the member's width (an i64 on the wire into an int32_t member
46// is an error, never a silent truncation);
47// * unknown fields are skipped structurally (forward compatibility) under the same limits.

193template <class M>
194struct message_info {
195 static constexpr std::size_t n = std::tuple_size_v<fields_tuple_t<M>>;
196 static_assert(n <= 64, "nanom: a tagged message may have at most 64 fields");
197
198 static consteval std::array<std::uint16_t, n> make_ids() {
199 std::array<std::uint16_t, n> a{};
200 std::size_t i = 0;
201 for_each_field<M>([&](auto f) { a[i++] = member_t<decltype(f)::mem_ptr>::id; });
202 return a;
203 }
204 static constexpr std::array<std::uint16_t, n> ids = make_ids();
205
206 static consteval bool ascending() {
207 for (std::size_t i = 1; i < n; ++i)
208 if (ids[i] <= ids[i - 1]) return false;
209 return true;
210 }
211 static_assert(ascending(),
212 "nanom: tagged message fields must be declared (and registered) with strictly "
213 "ascending, unique ids");
214
215 static consteval std::uint64_t make_required_mask() {
216 std::uint64_t m = 0;
217 std::size_t i = 0;
218 for_each_field<M>([&](auto f) {
219 using F = member_t<decltype(f)::mem_ptr>;
220 if (F::pres == presence::required && !is_optional_t<typename F::value_type>::value)
221 m |= std::uint64_t(1) << i;
222 ++i;
223 });
224 return m;
225 }
226 static constexpr std::uint64_t required_mask = make_required_mask();
228 static constexpr bool has_unknown = n > 0 && ids[0] == 0;
229
230 static constexpr std::uint16_t max_id = n ? ids[n - 1] : 0;
231 static constexpr bool dense = max_id < 256;
232 static constexpr std::uint8_t no_slot = 0xff;
233 static consteval auto make_slots() {
234 std::array<std::uint8_t, dense ? std::size_t(max_id) + 1 : 1> s{};
235 for (auto& x : s) x = no_slot;
236 if constexpr (dense)
237 for (std::size_t i = 0; i < n; ++i) s[ids[i]] = std::uint8_t(i);
238 return s;
239 }
240 static constexpr auto slots = make_slots();
241
243 static constexpr std::uint8_t slot_of(std::int32_t id) {
244 if (id < 0) return no_slot;
245 if constexpr (dense) {
246 return std::size_t(id) < slots.size() ? slots[std::size_t(id)] : no_slot;
247 } else {
248 for (std::size_t i = 0; i < n; ++i) // sorted ids: binary search is no faster at n <= 64
249 if (ids[i] == id) return std::uint8_t(i);
250 return no_slot;
251 }
252 }
253};

352// ---------------------------------------------------------------------------
353// 25. Thrift compact protocol — decode + encode
354// ---------------------------------------------------------------------------
355//
356// Wire summary (Apache Thrift compact protocol spec):
357// field header : (delta << 4) | type when the id delta is 1..15, else type byte + zigzag-varint i16 id
358// bool field : value IS the header type (1 = true, 2 = false); in a list it is one byte
359// i8 : one byte; i16/i32/i64 : zigzag varint; double : 8 bytes little-endian
360// binary/string: varint length + bytes
361// list/set : (size << 4) | elem_type when size < 15, else 0xF0 | elem_type + varint size
362// map : varint size, then (key_type << 4) | value_type when size > 0
363// struct : fields until a STOP (0x00) byte
364

367 stop = 0, bool_true = 1, bool_false = 2, i8 = 3, i16 = 4, i32 = 5, i64 = 6, dbl = 7,
368 binary = 8, list = 9, set = 10, map = 11, strct = 12
369};
370} // namespace thrift

379template <class T>
380consteval std::uint8_t type_of() {
381 if constexpr (is_optional_t<T>::value) return type_of<typename T::value_type>();
382 else if constexpr (std::is_same_v<T, bool>) return std::uint8_t(ctype::bool_true);
383 else if constexpr (std::is_same_v<T, std::int8_t>) return std::uint8_t(ctype::i8);
384 else if constexpr (std::is_same_v<T, std::int16_t>) return std::uint8_t(ctype::i16);
385 else if constexpr (std::is_same_v<T, std::int32_t>) return std::uint8_t(ctype::i32);
386 else if constexpr (std::is_same_v<T, std::int64_t>) return std::uint8_t(ctype::i64);
387 else if constexpr (std::is_same_v<T, double>) return std::uint8_t(ctype::dbl);
388 else if constexpr (std::is_enum_v<T>) {
389 static_assert(sizeof(T) == 4 && std::is_signed_v<std::underlying_type_t<T>>,
390 "nanom thrift: enums travel as i32 — give the enum an int32_t underlying type");
391 return std::uint8_t(ctype::i32);
392 }
393 else if constexpr (std::is_same_v<T, std::string_view> || std::is_same_v<T, std::string> ||
394 std::is_same_v<T, bytes>)
395 return std::uint8_t(ctype::binary);
396 else if constexpr (is_vector_t<T>::value || is_list_t<T>::value) return std::uint8_t(ctype::list);
397 else if constexpr (Message<T> || is_lazy_t<T>::value || std::is_same_v<T, empty_struct>)
398 return std::uint8_t(ctype::strct);
399 else static_assert(always_false<T>,
400 "nanom thrift: unsupported member type (Thrift has no unsigned integers or "
401 "float; use int8/16/32/64_t, double, bool, int32 enums, string_view/string/"
402 "bytes, nm::list/std::vector, nested messages, nm::lazy, nm::empty_struct)");
403}

419// ---- decoder ------------------------------------------------------------------------------------
420//
421// The hot loops run on raw [p, e) byte pointers with ONE fault slot per decode (ctx), instead of
422// threading an expected<done<T>, error> through every byte: a Thrift footer is millions of tiny
423// values, and the per-value result plumbing dominated the cost. Nothing here is less checked —
424// every read is bounds-checked against e, every count and length against the bytes left — the
425// error is just built once, at the API boundary (ctx::to_error), from the recorded fault.
426
427enum class fault : std::uint8_t {
428 none, truncated, bad_varint, out_of_range, bad_type, type_mismatch, too_deep, bad_count,
429 bad_bool, bad_id, missing_required
430};
431
432struct ctx {
433 input origin;
434 const std::byte* at = nullptr;
435 fault f = fault::none;
436 std::uint32_t need = 0;
437 const char* where = nullptr;
438
439 constexpr bool fail(fault k, const std::byte* p, std::size_t needed = 0) {
440 if (f == fault::none) {
441 f = k;
442 at = p;
443 need = std::uint32_t(std::min<std::size_t>(needed, max_incomplete_needed));
444 }
445 return false;
446 }
447 constexpr error to_error() const {
448 error e;
449 e.offset = std::uint32_t(at && origin.base ? std::size_t(at - origin.base) : origin.offset());
450 switch (f) {
451 case fault::truncated:
452 e.kind = origin.live ? errk::incomplete : errk::err;
453 e.expected = "more input";
454 e.needed = need;
455 break;
456 case fault::bad_varint: e.expected = "varint fitting the target width"; break;
457 case fault::out_of_range: e.expected = "integer within the member's range"; break;
458 case fault::bad_type: e.expected = "a valid thrift compact type"; break;
459 case fault::type_mismatch: e.expected = "thrift wire type matching the member's type"; break;
460 case fault::too_deep: e.expected = "thrift nesting within the depth limit"; break;
461 case fault::bad_count: e.expected = "thrift list/map count that fits the remaining bytes"; break;
462 case fault::bad_bool: e.expected = "thrift bool element (0, 1 or 2)"; break;
463 case fault::bad_id: e.expected = "thrift field id within i16"; break;
464 case fault::missing_required: e.expected = "every required thrift field"; break;
465 case fault::none: e.expected = "(no fault)"; break;
466 }
467 if (f == fault::missing_required && where) e.push_context(where, e.offset);
468 return e;
469 }
470};

474constexpr std::uint8_t u8at(ptr p) { return std::uint8_t(*p); }

477constexpr bool varint(ctx& c, ptr& p, ptr e, std::uint64_t& v) {
478 const ptr s = p;
479 if (e - p >= 10) {
480 std::uint64_t b = u8at(p);
481 if (b < 0x80) { v = b; p += 1; return true; }
482 std::uint64_t r = b & 0x7f;
483 for (int i = 1; i < 10; ++i) {
484 b = u8at(p + i);
485 r |= (b & 0x7f) << (7 * i);
486 if (b < 0x80) {
487 if (i == 9 && b > 1) return c.fail(fault::bad_varint, s);
488 v = r;
489 p += i + 1;
490 return true;
491 }
492 }
493 return c.fail(fault::bad_varint, s);
494 }
495 std::uint64_t r = 0;
496 for (int i = 0; i < 10; ++i) {
497 if (p + i >= e) return c.fail(fault::truncated, e, 1);
498 const std::uint64_t b = u8at(p + i);
499 r |= (b & 0x7f) << (7 * i);
500 if (b < 0x80) {
501 if (i == 9 && b > 1) return c.fail(fault::bad_varint, s);
502 v = r;
503 p += i + 1;
504 return true;
505 }
506 }
507 return c.fail(fault::bad_varint, s);
508}

511constexpr bool skip_varint(ctx& c, ptr& p, ptr e) {
512 const ptr s = p;
513 const ptr lim = e - p >= 10 ? p + 10 : e;
514 for (ptr q = p; q < lim; ++q) {
515 if (u8at(q) < 0x80) {
516 if (q - s == 9 && u8at(q) > 1) return c.fail(fault::bad_varint, s);
517 p = q + 1;
518 return true;
519 }
520 }
521 return lim == e && e - s < 10 ? c.fail(fault::truncated, e, 1) : c.fail(fault::bad_varint, s);
522}

524constexpr bool varint32(ctx& c, ptr& p, ptr e, std::uint32_t& v) {
525 const ptr s = p;
526 std::uint64_t w = 0;
527 if (!varint(c, p, e, w)) return false;
528 if (w > 0xffffffffu) return c.fail(fault::bad_varint, s);
529 v = std::uint32_t(w);
530 return true;
531}

533template <std::signed_integral S>
534constexpr bool zigzag(ctx& c, ptr& p, ptr e, S& v) {
535 const ptr s = p;
536 std::uint64_t w = 0;
537 if (!varint(c, p, e, w)) return false;
538 const std::int64_t d = zigzag_decode(w);
539 if (d < std::numeric_limits<S>::min() || d > std::numeric_limits<S>::max())
540 return c.fail(fault::out_of_range, s);
541 v = S(d);
542 return true;
543}

545struct list_header { std::uint32_t count; std::uint8_t elem; };
546constexpr bool read_list_header(ctx& c, ptr& p, ptr e, list_header& h) {
547 const ptr s = p;
548 if (p >= e) return c.fail(fault::truncated, p, 1);
549 const std::uint8_t b = u8at(p++);
550 h.elem = b & 0x0f;
551 h.count = b >> 4;
552 if (h.count == 15 && !varint32(c, p, e, h.count)) return false;
553 if (h.count > std::size_t(e - p) / min_elem_bytes(h.elem)) return c.fail(fault::bad_count, s);
554 return true;
555}

561constexpr bool skip_elems(ctx& c, ptr& p, ptr e, list_header h, int depth) {
562 switch (ctype(h.elem)) { // fixed-size / scalar elements: no per-element recursion
563 case ctype::bool_true: case ctype::bool_false: case ctype::i8:
564 p += h.count; // read_list_header proved count <= bytes left
565 return true;
566 case ctype::dbl:
567 p += std::size_t(h.count) * 8; // ... and count * 8 <= bytes left for doubles
568 return true;
569 case ctype::i16: case ctype::i32: case ctype::i64:
570 for (std::uint32_t i = 0; i < h.count; ++i)
571 if (!skip_varint(c, p, e)) return false;
572 return true;
573 default:
574 for (std::uint32_t i = 0; i < h.count; ++i)
575 if (!skip(c, p, e, h.elem, true, depth + 1)) return false;
576 return true;
577 }

580constexpr bool skip_struct(ctx& c, ptr& p, ptr e, int depth) {
581 if (depth > max_tagged_depth) return c.fail(fault::too_deep, p);
582 for (;;) {
583 if (p >= e) return c.fail(fault::truncated, p, 1);
584 const std::uint8_t h = u8at(p++);
585 if (h == 0) return true;
586 if ((h >> 4) == 0 && !skip_varint(c, p, e)) return false; // long-form id: parse, ignore
587 // scalars inline (the common case); containers and structs recurse
588 switch (ctype(h & 0x0f)) {
589 case ctype::bool_true: case ctype::bool_false:
590 continue;
591 case ctype::i8:
592 if (p >= e) return c.fail(fault::truncated, p, 1);
593 ++p;
594 continue;
595 case ctype::i16: case ctype::i32: case ctype::i64:
596 if (!skip_varint(c, p, e)) return false;
597 continue;
598 case ctype::binary: {
599 std::uint32_t n = 0;
600 if (!varint32(c, p, e, n)) return false;
601 if (std::size_t(e - p) < n) return c.fail(fault::truncated, e, n - std::size_t(e - p));
602 p += n;
603 continue;
604 }
605 default:
606 if (!skip(c, p, e, h & 0x0f, false, depth)) return false;
607 }
608 }
609}

613constexpr bool skip(ctx& c, ptr& p, ptr e, std::uint8_t type, bool in_container, int depth) {
614 switch (ctype(type)) {
615 case ctype::bool_true:
616 case ctype::bool_false:
617 if (!in_container) return true;
618 [[fallthrough]];
619 case ctype::i8:
620 if (p >= e) return c.fail(fault::truncated, p, 1);
621 ++p;
622 return true;
623 case ctype::i16: case ctype::i32: case ctype::i64:
624 return skip_varint(c, p, e);
625 case ctype::dbl:
626 if (e - p < 8) return c.fail(fault::truncated, e, std::size_t(8 - (e - p)));
627 p += 8;
628 return true;
629 case ctype::binary: {
630 std::uint32_t n = 0;
631 if (!varint32(c, p, e, n)) return false;
632 if (std::size_t(e - p) < n) return c.fail(fault::truncated, e, n - std::size_t(e - p));
633 p += n;
634 return true;
635 }
636 case ctype::list: case ctype::set: {
637 if (depth > max_tagged_depth) return c.fail(fault::too_deep, p);
638 list_header h{};
639 return read_list_header(c, p, e, h) && skip_elems(c, p, e, h, depth);
640 }
641 case ctype::map: {
642 if (depth > max_tagged_depth) return c.fail(fault::too_deep, p);
643 const ptr s = p;
644 std::uint32_t n = 0;
645 if (!varint32(c, p, e, n)) return false;
646 if (n == 0) return true;
647 if (p >= e) return c.fail(fault::truncated, p, 1);
648 const std::uint8_t kv = u8at(p++);
649 const std::uint8_t kt = kv >> 4, vt = kv & 0x0f;
650 if (n > std::size_t(e - p) / (min_elem_bytes(kt) + min_elem_bytes(vt)))
651 return c.fail(fault::bad_count, s);
652 for (std::uint32_t i = 0; i < n; ++i)
653 if (!skip(c, p, e, kt, true, depth + 1) || !skip(c, p, e, vt, true, depth + 1)) return false;
654 return true;
655 }
656 case ctype::strct:
657 return skip_struct(c, p, e, depth + 1);
658 case ctype::stop:
659 break;
660 }
661 return c.fail(fault::bad_type, p);
662}

668template <class T>
669constexpr bool read_value(ctx& c, ptr& p, ptr e, int depth, T& out) {
670 if constexpr (std::is_same_v<T, bool>) { // container element; bool fields are handled inline
671 if (p >= e) return c.fail(fault::truncated, p, 1);
672 const std::uint8_t b = u8at(p);
673 if (b > 2) return c.fail(fault::bad_bool, p);
674 ++p;
675 out = b == 1;
676 return true;
677 } else if constexpr (std::is_same_v<T, std::int8_t>) {
678 if (p >= e) return c.fail(fault::truncated, p, 1);
679 out = std::bit_cast<std::int8_t>(u8at(p++));
680 return true;
681 } else if constexpr (std::is_same_v<T, std::int16_t> || std::is_same_v<T, std::int32_t> ||
682 std::is_same_v<T, std::int64_t>) {
683 return zigzag(c, p, e, out);
684 } else if constexpr (std::is_same_v<T, double>) {
685 if (e - p < 8) return c.fail(fault::truncated, e, std::size_t(8 - (e - p)));
686 std::uint64_t u = 0;
687 for (int i = 0; i < 8; ++i) u |= std::uint64_t(u8at(p + i)) << (8 * i);
688 p += 8;
689 out = std::bit_cast<double>(u);
690 return true;
691 } else if constexpr (std::is_enum_v<T>) {
692 std::int32_t v = 0;
693 if (!zigzag(c, p, e, v)) return false;
694 out = T(v);
695 return true;
696 } else if constexpr (std::is_same_v<T, std::string_view> || std::is_same_v<T, std::string> ||
697 std::is_same_v<T, bytes>) {
698 std::uint32_t n = 0;
699 if (!varint32(c, p, e, n)) return false;
700 if (std::size_t(e - p) < n) return c.fail(fault::truncated, e, n - std::size_t(e - p));
701 const bytes b = c.origin.attested_span(p, n);
702 p += n;
703 if constexpr (std::is_same_v<T, bytes>) out = b;
704 else out = T(as_str(b));
705 return true;
706 } else if constexpr (is_vector_t<T>::value) {
707 using E = typename T::value_type;
708 const ptr s = p;
709 list_header h{};
710 if (!read_list_header(c, p, e, h)) return false;
711 if (h.count && !wire_ok(type_of<E>(), h.elem)) return c.fail(fault::type_mismatch, s);
712 if (depth > max_tagged_depth) return c.fail(fault::too_deep, s);
713 out.clear();
714 out.reserve(h.count); // bounded: read_list_header checked count against the bytes left
715 for (std::uint32_t i = 0; i < h.count; ++i) {
716 E v{};
717 if (!read_value<E>(c, p, e, depth + 1, v)) return false;
718 out.push_back(std::move(v));
719 }
720 return true;
721 } else if constexpr (is_list_t<T>::value) {
722 using E = typename T::value_type;
723 const ptr s = p;
724 list_header h{};
725 if (!read_list_header(c, p, e, h)) return false;
726 if (h.count && !wire_ok(type_of<E>(), h.elem)) return c.fail(fault::type_mismatch, s);
727 if (depth > max_tagged_depth) return c.fail(fault::too_deep, s);
728 const ptr body = p;
729 // structural validation finds the end (Thrift lists carry no byte length); the elements are
730 // decoded later, on iteration
731 if (!skip_elems(c, p, e, h, depth)) return false;
732 out = T(c.origin.with_range(body, p), h.count, h.elem, std::uint8_t(depth + 1));
733 return true;
734 } else if constexpr (is_lazy_t<T>::value) {
735 const ptr s = p;
736 if (!skip_struct(c, p, e, depth + 1)) return false;
737 out = T(c.origin.with_range(s, p), std::uint8_t(depth + 1));
738 return true;
739 } else if constexpr (std::is_same_v<T, empty_struct>) {
740 return skip_struct(c, p, e, depth + 1);
741 } else if constexpr (Message<T>) {
742 return read_struct<T>(c, p, e, depth + 1, out);
743 } else {
744 static_assert(always_false<T>, "nanom thrift: unsupported member type");
745 }
746}

748template <Message M>
749constexpr bool read_struct(ctx& c, ptr& p, ptr e, int depth, M& m) {
750 using info = message_info<M>;
751 static_assert(!info::has_unknown, "nanom thrift: pb_unknown is protobuf-only");
752 if (depth > max_tagged_depth) return c.fail(fault::too_deep, p);
753 const ptr start = p;
754 std::uint64_t seen = 0;
755 std::int32_t last = 0;
756 for (;;) {
757 if (p >= e) return c.fail(fault::truncated, p, 1);
758 const ptr hp = p;
759 const std::uint8_t hb = u8at(p++);
760 if (hb == 0) break; // STOP
761 const std::uint8_t type = hb & 0x0f, delta = hb >> 4;
762 if (delta == 0) {
763 std::int16_t id = 0;
764 if (!zigzag(c, p, e, id)) return false;
765 last = id;
766 } else {
767 last += delta;
768 if (last > 32767) return c.fail(fault::bad_id, hp);
769 }
770 const std::uint8_t slot = info::slot_of(last);
771 if (slot == info::no_slot) { // unknown field: skip, stay forward compatible
772 if (!skip(c, p, e, type, false, depth)) return false;
773 continue;
774 }
775 bool matched = true;
776 const bool ok = with_field_index<M>(slot, [&]<std::size_t I>(std::integral_constant<std::size_t, I>) {
777 using F = field_at_t<M, I>;
778 using V = typename F::value_type;
779 using T = unwrap_optional_t<V>;
780 constexpr auto mp = std::tuple_element_t<I, fields_tuple_t<M>>::mem_ptr;
781 if (!wire_ok(type_of<T>(), type)) { // Thrift semantics: skip, never reinterpret
782 matched = false;
783 return skip(c, p, e, type, false, depth);
784 }
785 if constexpr (std::is_same_v<T, bool>) { // bool field: the value is the header type
786 (m.*mp).v = V(type == std::uint8_t(ctype::bool_true));
787 return true;
788 } else if constexpr (is_optional_t<V>::value) {
789 auto& slot_v = (m.*mp).v;
790 if (!slot_v) slot_v.emplace();
791 return read_value<T>(c, p, e, depth, *slot_v);
792 } else {
793 return read_value<T>(c, p, e, depth, (m.*mp).v);
794 }
795 });
796 if (!ok) return false;
797 if (matched) seen |= std::uint64_t(1) << slot;
798 }
799 if ((seen & info::required_mask) != info::required_mask) {
800 c.where = describe<M>::name();
801 return c.fail(fault::missing_required, start);
802 }
803 return true;
804}
