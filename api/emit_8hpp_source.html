2// nanom/emit.hpp — WRITE fixed-layout described structs: the inverse of strct<T>() / overlay<T>().
3//
4// The same description that parses a header writes it, field by field and bit by bit:
5//
6// struct udp_hdr { nm::be<u16> src_port, dst_port, length, checksum; };
7// NANOM_DESCRIBE(udp_hdr, src_port, dst_port, length, checksum);
8//
9// auto bytes = nm::to_bytes(hdr); // std::array<std::byte, 8>, constexpr
10// auto n = nm::emit(hdr, out_span); // into a buffer (never written past)
11// auto m = nm::emit_frame(hdr, payload, sink); // header + payload into any byte_sink
12//
13// Fields are written exactly as strct<T>() reads them: be<> / le<> keep their wire bytes, plain
14// scalars use the `dflt` byte order, ubits / ibits are packed msb0 or lsb0 at their compile-time
15// bit offsets, arrays and nested described structs recurse. So strct(emit(x)) == x, and for any
16// byte string b of the right size, emit(strct(b)) == b.
17//
18// Computed fields (binrw's `calc`): specialize nm::computed<T> to derive fields while writing —
19// lengths from the payload, checksums over the encoded bytes:
20//
21// template <> struct nm::computed<udp_hdr> {
22// static constexpr auto fields = std::tuple{
23// nm::calc<"length">([](const udp_hdr&, const nm::emit_ctx& c) { return 8 + c.payload.size(); }),
24// nm::checksum<"checksum">([](std::span<const std::byte> hdr, const nm::emit_ctx& c) { ... }),
25// };
26// };
27//
28// calc<"f">(fn(const T&, const emit_ctx&)) runs before encoding (in order);
29// checksum<"f">(fn(bytes, const emit_ctx&)) runs after, over the encoded struct with field
30// f zeroed, then f is written in place.
31// A computed value is range-checked into its field (a length that does not fit a be<u16> is an
32// error, never a truncation). verify_computed(bytes, ctx) checks stored values on the read side.
33//
34// Errors (encode_error, see nanom/sink.hpp): a ubits / ibits value outside its N bits, a computed
35// value that does not fit its field, an output buffer smaller than wire_size_v<T>, a sink that
36// refuses bytes. Reading code never includes this header.

87// ---- bit writing: the exact inverse of read_bits ------------------------------------------------
88constexpr void write_bits(std::byte* p, std::size_t bitoff, unsigned n, std::uint64_t v, bit_order ord) {
89 std::byte* q = p + bitoff / 8;
90 std::size_t bit = bitoff % 8;
91 for (unsigned got = 0; got < n;) {
92 const unsigned avail = unsigned(8 - bit);
93 const unsigned use = std::min(avail, n - got);
94 const unsigned mask = (1u << use) - 1;
95 unsigned piece, shift;
96 if (ord == bit_order::msb0) {
97 piece = unsigned(v >> (n - got - use)) & mask; // the next `use` bits from the top
98 shift = avail - use;
99 } else {
100 piece = unsigned(v >> got) & mask; // the next `use` bits from the bottom
101 shift = unsigned(bit);
102 }
103 const unsigned keep = ~(mask << shift) & 0xffu;
104 *q = std::byte((unsigned(std::uint8_t(*q)) & keep) | (piece << shift));
105 got += use;
106 bit += use;
107 if (bit == 8) {
108 bit = 0;
109 ++q;
110 }
111 }
112}

114template <class U>
115constexpr void store_uint(std::byte* q, U u, std::size_t nbytes, std::endian order) {
116 for (std::size_t i = 0; i < nbytes; ++i) {
117 const std::size_t sh = order == std::endian::big ? 8 * (nbytes - 1 - i) : 8 * i;
118 q[i] = std::byte(std::uint8_t(u >> sh));
119 }
120}

123template <class F>
124constexpr const char* emit_field(const F& v, std::byte* p, std::size_t bitoff, std::endian dflt) {
125 if constexpr (requires { v.raw; typename F::value_type; F::order; }) { // be / le: wire bytes
126 for (std::size_t i = 0; i < v.raw.size(); ++i) p[bitoff / 8 + i] = v.raw[i];
127 return nullptr;
128 } else if constexpr (requires { F::order; F::bits; }) { // ubits / ibits
129 using VT = typename F::value_type;
130 constexpr unsigned N = F::bits;
131 std::uint64_t raw;
132 if constexpr (std::is_signed_v<VT>) {
133 const std::int64_t s = std::int64_t(v.v);
134 raw = std::uint64_t(s);
135 if constexpr (N < 64) {
136 constexpr std::int64_t lo = -(std::int64_t(1) << (N - 1)), hi = (std::int64_t(1) << (N - 1)) - 1;
137 if (s < lo || s > hi) return "a value outside its signed bit field";
138 raw &= (std::uint64_t(1) << N) - 1;
139 }
140 } else {
141 raw = std::uint64_t(v.v);
142 if constexpr (N < 64)
143 if ((raw >> N) != 0) return "a value wider than its bit field";
144 }
145 write_bits(p, bitoff, N, raw, F::order);
146 return nullptr;
147 } else if constexpr (is_std_array_v<F>) {
148 using E = typename F::value_type;
149 constexpr std::size_t eb = wire<E>::bits;
150 for (std::size_t i = 0; i < v.size(); ++i)
151 if (const char* e = emit_field<E>(v[i], p, bitoff + i * eb, dflt)) return e;
152 return nullptr;
153 } else if constexpr (std::integral<F> || std::floating_point<F>) {
154 using U = uint_for_bytes<sizeof(F)>;
155 store_uint(p + bitoff / 8, std::bit_cast<U>(v), sizeof(F), dflt);
156 return nullptr;
157 } else { // nested described struct
158 constexpr auto offs = field_bit_offsets<F>();
159 const char* err = nullptr;
160 std::size_t i = 0;
161 for_each_field<F>([&](auto f) {
162 const std::size_t o = offs[i++];
163 if (!err) err = emit_field<member_t<decltype(f)::mem_ptr>>(v.*(decltype(f)::mem_ptr), p, bitoff + o, dflt);
164 });
165 return err;
166 }
167}

170template <class T>
171constexpr std::string_view first_bad_field(const T& v, std::endian dflt) {
172 std::array<std::byte, wire_size_v<T>> scratch{};
173 constexpr auto offs = field_bit_offsets<T>();
174 std::string_view bad;
175 std::size_t i = 0;
176 for_each_field<T>([&](auto f) {
177 const std::size_t o = offs[i++];
178 if (bad.empty() && emit_field<member_t<decltype(f)::mem_ptr>>(v.*(decltype(f)::mem_ptr), scratch.data(), o, dflt))
179 bad = decltype(f)::name.sv();
180 });
181 return bad;
182}

185template <class F, class V>
186constexpr bool set_computed(F& field, V value) {
187 if constexpr (requires { field.raw; typename F::value_type; F::order; }) { // be / le
188 using T = typename F::value_type;
189 if (!std::in_range<T>(value)) return false;
190 field.set(T(value));
191 return true;
192 } else if constexpr (requires { F::order; F::bits; }) { // ubits / ibits
193 using VT = typename F::value_type;
194 constexpr unsigned N = F::bits;
195 if constexpr (std::is_signed_v<VT>) {
196 if (!std::in_range<std::int64_t>(value)) return false;
197 if constexpr (N < 64) {
198 const auto s = std::int64_t(value);
199 if (s < -(std::int64_t(1) << (N - 1)) || s > (std::int64_t(1) << (N - 1)) - 1) return false;
200 }
201 } else {
202 if (!std::in_range<std::uint64_t>(value)) return false;
203 if constexpr (N < 64)
204 if ((std::uint64_t(value) >> N) != 0) return false;
205 }
206 field.v = VT(value);
207 return true;
208 } else if constexpr (std::integral<F>) {
209 if (!std::in_range<F>(value)) return false;
210 field = F(value);
211 return true;
212 } else {
213 static_assert(sizeof(F) == 0, "nanom: a computed field must be an integer, be<>/le<> or bit field");
214 }
215}

217template <class T, fixed_string Name>
218constexpr auto& field_ref(T& v) {
219 constexpr std::size_t I = field_index<T, Name>();
220 static_assert(I != std::size_t(-1), "nanom: computed<T> names a field T does not have");
221 using FD = std::remove_cvref_t<decltype(std::get<I>(describe<T>::fields()))>;
222 return v.*(FD::mem_ptr);

226template <Described T>
227constexpr expected<std::size_t, encode_error> emit_struct(const T& value, std::byte* p, std::endian dflt,
228 const emit_ctx& ctx) {
229 static_assert(layout_ok<T>(),
230 "nanom: bit fields must pack to byte boundaries, every non-bit field must start byte-aligned, "
231 "and msb0 / lsb0 runs must not share a byte");
232 constexpr std::size_t size = wire_size_v<T>;
233 constexpr auto offs = field_bit_offsets<T>();
234 T v = value;
235 const auto& specs = computed<T>::fields;
236 // 1. calc fields, in order
237 std::string_view bad;
238 std::apply(
239 [&](const auto&... spec) {
240 (
241 [&] {
242 using S = std::remove_cvref_t<decltype(spec)>;
243 if constexpr (is_calc<S>::value) {
244 if (bad.empty() && !set_computed(field_ref<T, S::name>(v), spec.fn(std::as_const(v), ctx)))
245 bad = S::name.sv();
246 }
247 }(),
248 ...);
249 },
250 specs);
251 if (!bad.empty())
252 return unexpected<encode_error>(encode_error{"a computed value that does not fit its field", describe<T>::name(), bad});
253 // 2. every field
254 for (std::size_t k = 0; k < size; ++k) p[k] = std::byte{0};
255 {
256 const char* err = nullptr;
257 std::size_t i = 0;
258 for_each_field<T>([&](auto f) {
259 const std::size_t o = offs[i++];
260 if (!err) err = emit_field<member_t<decltype(f)::mem_ptr>>(v.*(decltype(f)::mem_ptr), p, o, dflt);
261 });
262 if (err) return unexpected<encode_error>(encode_error{err, describe<T>::name(), first_bad_field(v, dflt)});
263 }
264 // 3. checksum fields: computed over the encoding with the field zeroed, then written in place
265 std::apply(
266 [&](const auto&... spec) {
267 (
268 [&] {
269 using S = std::remove_cvref_t<decltype(spec)>;
270 if constexpr (is_checksum<S>::value) {
271 if (!bad.empty()) return;
272 constexpr std::size_t I = field_index<T, S::name>();
273 static_assert(I != std::size_t(-1), "nanom: computed<T> names a field T does not have");
274 using F = field_type_at<T, I>;
275 auto& fld = field_ref<T, S::name>(v);
276 F zero{};
277 (void)emit_field<F>(zero, p, offs[I], dflt);
278 if (!set_computed(fld, spec.fn(std::span<const std::byte>(p, size), ctx)) ||
279 emit_field<F>(fld, p, offs[I], dflt))
280 bad = S::name.sv();
281 }
282 }(),
283 ...);
284 },
285 specs);
286 if (!bad.empty())
287 return unexpected<encode_error>(encode_error{"a computed value that does not fit its field", describe<T>::name(), bad});
288 return size;
289}

317template <Described T, byte_sink S>
318expected<std::size_t, encode_error> emit_to(const T& v, S& sink, const emit_ctx& ctx = {},
319 std::endian dflt = std::endian::native) {
320 auto b = to_bytes(v, ctx, dflt);
321 if (!b) return unexpected<encode_error>(b.error());
322 if (!sink.put(b->data(), b->size()))
323 return unexpected<encode_error>(encode_error{"the sink refused the bytes (full or failed)", describe<T>::name(), {}});
324 return b->size();
325}

329template <Described T, byte_sink S>
330expected<std::size_t, encode_error> emit_frame(const T& v, std::span<const std::byte> payload, S& sink,
331 std::endian dflt = std::endian::native) {
332 auto n = emit_to(v, sink, emit_ctx{payload}, dflt);
333 if (!n) return n;
334 if (!payload.empty() && !sink.put(payload.data(), payload.size()))
335 return unexpected<encode_error>(encode_error{"the sink refused the bytes (full or failed)", describe<T>::name(), {}});
336 return *n + payload.size();
337}

341template <Described T>
342constexpr bool verify_computed(std::span<const std::byte> wire, const emit_ctx& ctx = {},
343 std::endian dflt = std::endian::native) {
344 if (wire.size() < wire_size_v<T>) return false;
345 auto v = strct<T>(dflt)(from(wire));
346 if (!v) return false;
347 auto again = to_bytes(v->value, ctx, dflt);
348 if (!again) return false;
349 for (std::size_t i = 0; i < wire_size_v<T>; ++i)
350 if ((*again)[i] != wire[i]) return false;
351 return true;
352}
