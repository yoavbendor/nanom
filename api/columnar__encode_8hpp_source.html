2// nanom/columnar_encode.hpp — WRITE columnar pages: the encoders of nanom/columnar.hpp's decoders.
3//
4// Every encoder here writes exactly what the matching decoder reads (tests/test_columnar_encode.cpp
5// round-trips each one through it, and fuzz/fuzz_columnar_encode.cpp fuzzes the pairs):
6//
7// pack_bits<U> LSB-first bit packing <-> unpack_bits<U>
8// rle_hybrid_encode RLE / bit-packed hybrid (levels, dictionary <-> rle_bp_decoder
9// indices, RLE booleans): runs of >= 8 equal values become RLE runs,
10// everything else bit-packed groups of 8
11// rle_bitmap_encode the same, straight from a bitmap (width 1: a <-> rle_bitmap
12// literal group IS one bitmap byte)
13// delta_binary_packed_encode<T> DELTA_BINARY_PACKED (int32 / int64) <-> delta_binary_packed<T>
14// byte_stream_split_encode BYTE_STREAM_SPLIT <-> byte_stream_split
15// delta_length_encode DELTA_LENGTH_BYTE_ARRAY <-> delta_length_views
16// delta_prefix_encode DELTA_BYTE_ARRAY (front coding) <-> delta_prefix_views
17// fixed_dictionary<T>, string_dictionary dictionary builders (open addressing, exact: values
18// are compared by their bytes, so -0.0 / +0.0 and NaN payloads stay apart)
19// value_stats / binary_stats min / max / null count in Parquet's orders
20//
21// Encoders append to a std::vector<std::byte> (the page body before compression). Inputs are the
22// caller's values; nothing is read past the spans given. Reading code never includes this header.

50template <class U>
51 requires(std::unsigned_integral<U>)
52inline void pack_bits(std::span<const U> in, unsigned width, std::vector<std::byte>& out) {
53 if (width == 0 || in.empty()) return;
54 const std::uint64_t mask = width >= 64 ? ~std::uint64_t(0) : (std::uint64_t(1) << width) - 1;
55 const std::size_t total = (in.size() * width + 7) / 8;
56 const std::size_t base = out.size();
57 out.resize(base + total + 8); // + slack for the whole-word stores
58 std::byte* o = out.data() + base;
59 std::uint64_t acc = 0;
60 unsigned bits = 0;
61 for (const U x : in) {
62 const std::uint64_t v = std::uint64_t(x) & mask;
63 acc |= v << bits;
64 if (bits + width >= 64) {
65 std::memcpy(o, &acc, 8); // little-endian hosts (static_assert'ed by callers' formats)
66 o += 8;
67 acc = bits ? v >> (64 - bits) : 0;
68 bits = bits + width - 64;
69 } else {
70 bits += width;
71 }
72 }
73 std::memcpy(o, &acc, 8);
74 out.resize(base + total);
75}

85template <class V>
86 requires(std::unsigned_integral<V>)
87inline void rle_hybrid_encode(std::span<const V> values, unsigned width, std::vector<std::byte>& out) {
88 const std::size_t n = values.size();
89 const std::size_t vbytes = (width + 7) / 8;
90 std::size_t lit_begin = 0, lit_end = 0; // pending literal values [lit_begin, lit_end)
91 std::vector<V> pad;
92 const auto flush_literal = [&](bool last) {
93 const std::size_t count = lit_end - lit_begin;
94 if (count == 0) return;
95 const std::size_t groups = (count + 7) / 8;
96 enc_detail::put_uleb(out, (std::uint64_t(groups) << 1) | 1);
97 if (count % 8 == 0 || !last) {
98 pack_bits<V>(values.subspan(lit_begin, count), width, out);
99 } else { // the stream's final, partial group: pad with zeros
100 pad.assign(values.begin() + std::ptrdiff_t(lit_begin), values.begin() + std::ptrdiff_t(lit_end));
101 pad.resize(groups * 8, V(0));
102 pack_bits<V>(std::span<const V>(pad), width, out);
103 }
104 lit_begin = lit_end;
105 };
106 std::size_t i = 0;
107 while (i < n) {
108 std::size_t j = i + 1;
109 while (j < n && values[j] == values[i]) ++j;
110 std::size_t run = j - i;
111 if (run >= 8) {
112 // finish the pending literal's last group with values from this run
113 const std::size_t pending = lit_end - lit_begin;
114 if (pending % 8) {
115 const std::size_t steal = 8 - pending % 8;
116 lit_end += steal;
117 i += steal;
118 run -= steal;
119 }
120 if (run >= 8) {
121 flush_literal(false);
122 enc_detail::put_uleb(out, std::uint64_t(run) << 1);
123 enc_detail::put_le(out, std::uint64_t(values[i]), vbytes);
124 lit_begin = lit_end = j;
125 } else {
126 lit_end = j;
127 }
128 } else {
129 if (lit_begin == lit_end) lit_begin = i;
130 lit_end = j;
131 }
132 i = j;
133 }
134 flush_literal(true);
135}

140inline void rle_bitmap_encode(const std::uint8_t* bm, std::size_t n, std::vector<std::byte>& out) {
141 const std::size_t full = n / 8; // whole bytes; the partial tail goes into the final literal
142 std::size_t lit_from = 0; // first byte of the pending literal
143 const auto flush_literal = [&](std::size_t to, bool with_tail) {
144 std::size_t groups = to - lit_from + (with_tail ? 1 : 0);
145 if (groups == 0) return;
146 enc_detail::put_uleb(out, (std::uint64_t(groups) << 1) | 1);
147 for (std::size_t b = lit_from; b < to; ++b) out.push_back(std::byte(bm[b]));
148 if (with_tail) out.push_back(std::byte(bm[full] & std::uint8_t((1u << (n % 8)) - 1)));
149 };
150 std::size_t b = 0;
151 while (b < full) {
152 const std::uint8_t v = bm[b];
153 if (v == 0x00 || v == 0xff) {
154 std::size_t e = b + 1;
155 while (e + 8 <= full) { // 8 bytes at a time
156 std::uint64_t w;
157 std::memcpy(&w, bm + e, 8);
158 if (w != (v ? ~std::uint64_t(0) : 0)) break;
159 e += 8;
160 }
161 while (e < full && bm[e] == v) ++e;
162 if (e - b >= 2) {
163 flush_literal(b, false);
164 enc_detail::put_uleb(out, std::uint64_t(e - b) * 8 << 1);
165 out.push_back(std::byte(v ? 1 : 0));
166 lit_from = e;
167 b = e;
168 continue;
169 }
170 }
171 ++b;
172 }
173 flush_literal(full, n % 8 != 0);
174}

188template <class T>
189 requires(std::same_as<T, std::int32_t> || std::same_as<T, std::int64_t>)
190inline void delta_binary_packed_encode(std::span<const T> values, std::vector<std::byte>& out) {
191 using U = std::make_unsigned_t<T>;
192 constexpr std::size_t block = 128, minis = 4, per_mini = block / minis;
193 enc_detail::put_uleb(out, block);
194 enc_detail::put_uleb(out, minis);
195 enc_detail::put_uleb(out, values.size());
196 enc_detail::put_uleb(out, zigzag_encode(std::int64_t(values.empty() ? 0 : values[0])));
197 if (values.size() <= 1) return;
198 U deltas[block];
199 U adjusted[block];
200 for (std::size_t start = 1; start < values.size(); start += block) {
201 const std::size_t count = std::min(block, values.size() - start);
202 T min_delta = std::numeric_limits<T>::max();
203 for (std::size_t k = 0; k < count; ++k) {
204 deltas[k] = U(U(values[start + k]) - U(values[start + k - 1]));
205 min_delta = std::min(min_delta, std::bit_cast<T>(deltas[k]));
206 }
207 for (std::size_t k = 0; k < count; ++k) adjusted[k] = U(deltas[k] - U(min_delta));
208 for (std::size_t k = count; k < block; ++k) adjusted[k] = 0; // padding of the last miniblock
209 enc_detail::put_uleb(out, zigzag_encode(std::int64_t(min_delta)));
210 std::uint8_t widths[minis];
211 const std::size_t used = (count + per_mini - 1) / per_mini; // miniblocks holding values
212 for (std::size_t m = 0; m < minis; ++m) {
213 U hi = 0;
214 if (m < used)
215 for (std::size_t k = m * per_mini; k < (m + 1) * per_mini; ++k) hi |= adjusted[k];
216 widths[m] = std::uint8_t(std::bit_width(hi));
217 out.push_back(std::byte(widths[m]));
218 }
219 for (std::size_t m = 0; m < used; ++m) // unused miniblocks: width 0, no bytes
220 pack_bits<U>(std::span<const U>(adjusted + m * per_mini, per_mini), widths[m], out);
221 }

230inline bool byte_stream_split_encode(std::span<const std::byte> in, std::size_t width, std::size_t n,
231 std::span<std::byte> out) {
232 if (width == 0 || n > std::numeric_limits<std::size_t>::max() / width) return false;
233 if (in.size() < n * width || out.size() < n * width) return false;
234 std::byte* dst = out.data();
235 const std::byte* src = in.data();
236 if (width == 4) {
237 for (std::size_t i = 0; i < n; ++i) {
238 dst[i] = src[4 * i];
239 dst[n + i] = src[4 * i + 1];
240 dst[2 * n + i] = src[4 * i + 2];
241 dst[3 * n + i] = src[4 * i + 3];
242 }
243 } else {
244 for (std::size_t b = 0; b < width; ++b)
245 for (std::size_t i = 0; i < n; ++i) dst[b * n + i] = src[width * i + b];
246 }
247 return true;
248}

261inline bool delta_length_encode(std::span<const std::string_view> values, std::vector<std::byte>& out) {
262 std::vector<std::int32_t> lengths(values.size());
263 std::size_t total = 0;
264 for (std::size_t i = 0; i < values.size(); ++i) {
265 if (values[i].size() > std::size_t(INT32_MAX)) return false;
266 lengths[i] = std::int32_t(values[i].size());
267 total += values[i].size();
268 }
269 delta_binary_packed_encode<std::int32_t>(lengths, out);
270 out.reserve(out.size() + total);
271 for (const auto v : values) {
272 const auto* p = reinterpret_cast<const std::byte*>(v.data());
273 out.insert(out.end(), p, p + v.size());
274 }
275 return true;
276}

280inline bool delta_prefix_encode(std::span<const std::string_view> values, std::vector<std::byte>& out) {
281 std::vector<std::int32_t> prefix(values.size());
282 std::vector<std::string_view> suffix(values.size());
283 for (std::size_t i = 0; i < values.size(); ++i) {
284 if (values[i].size() > std::size_t(INT32_MAX)) return false;
285 std::size_t p = 0;
286 if (i) {
287 const std::size_t lim = std::min(values[i].size(), values[i - 1].size());
288 while (p < lim && values[i][p] == values[i - 1][p]) ++p;
289 }
290 prefix[i] = std::int32_t(p);
291 suffix[i] = values[i].substr(p);
292 }
293 delta_binary_packed_encode<std::int32_t>(prefix, out);
294 return delta_length_encode(suffix, out);
295}

302inline std::uint64_t load64(const unsigned char* p) {
303 std::uint64_t w;
304 std::memcpy(&w, p, 8);
305 return w;
306}
307#if defined(__SIZEOF_INT128__)
308__extension__ using uint128_t = unsigned __int128;
309inline std::uint64_t mix64(std::uint64_t a, std::uint64_t b) {
310 // 64x64 -> 128 multiply, folded (the wyhash / mum mix): one multiply per step
311 const uint128_t r = uint128_t(a) * b;
312 return std::uint64_t(r) ^ std::uint64_t(r >> 64);
313}
314#else
315inline std::uint64_t mix64(std::uint64_t a, std::uint64_t b) { // portable: the same folded product
316 const std::uint64_t al = a & 0xffffffffu, ah = a >> 32, bl = b & 0xffffffffu, bh = b >> 32;
317 const std::uint64_t ll = al * bl, lh = al * bh, hl = ah * bl, hh = ah * bh;
318 const std::uint64_t mid = (ll >> 32) + (lh & 0xffffffffu) + (hl & 0xffffffffu);
319 const std::uint64_t lo = (ll & 0xffffffffu) | (mid << 32);
320 const std::uint64_t hi = hh + (lh >> 32) + (hl >> 32) + (mid >> 32);
321 return lo ^ hi;
322}
323#endif
325inline std::uint64_t hash_bytes(const void* p, std::size_t n) {
326 constexpr std::uint64_t k0 = 0xa0761d6478bd642full, k1 = 0xe7037ed1a0b428dbull, k2 = 0x8ebc6af09c88c6e3ull;
327 const auto* b = static_cast<const unsigned char*>(p);
328 std::uint64_t lo, hi, seed = k0 ^ n;
329 if (n > 16) {
330 for (std::size_t i = 0; i + 16 < n; i += 16) seed = mix64(load64(b + i) ^ k1, load64(b + i + 8) ^ seed);
331 lo = load64(b + n - 16); // the last 16 bytes (overlapping what the loop saw)
332 hi = load64(b + n - 8);
333 } else if (n >= 8) {
334 lo = load64(b);
335 hi = load64(b + n - 8);
336 } else if (n >= 4) {
337 std::uint32_t x, y;
338 std::memcpy(&x, b, 4);
339 std::memcpy(&y, b + n - 4, 4);
340 lo = x;
341 hi = y;
342 } else {
343 lo = n ? (std::uint64_t(b[0]) << 16) | (std::uint64_t(b[n / 2]) << 8) | b[n - 1] : 0;
344 hi = 0;
345 }
346 return mix64(lo ^ k1, hi ^ seed ^ k2);
347}

351struct probe_slots {
352 std::vector<std::uint32_t> slots;
353 bool needs_grow(std::size_t entries) const { return (entries + 1) * 2 > slots.size(); }
354 template <class HashOf>
355 void grow(std::size_t, HashOf&& hash_of) {
356 std::vector<std::uint32_t> old;
357 old.swap(slots);
358 slots.assign(old.empty() ? 64 : old.size() * 2, 0u);
359 const std::size_t mask = slots.size() - 1;
360 for (const std::uint32_t at : old) {
361 if (!at) continue;
362 std::size_t s = std::size_t(hash_of(at - 1)) & mask;
363 while (slots[s]) s = (s + 1) & mask;
364 slots[s] = at;
365 }
366 }
367};
368} // namespace enc_detail

372template <class T>
373 requires(std::is_trivially_copyable_v<T>)
374class fixed_dictionary {
375 public:
377 std::uint32_t index_of(const T& v) {
378 if (t_.needs_grow(values_.size()))
379 t_.grow(values_.size(), [&](std::uint32_t i) { return hashes_[i]; });
380 const std::uint64_t h = enc_detail::hash_bytes(&v, sizeof(T));
381 const std::size_t mask = t_.slots.size() - 1;
382 for (std::size_t s = std::size_t(h) & mask;; s = (s + 1) & mask) {
383 const std::uint32_t at = t_.slots[s];
384 if (at == 0) {
385 t_.slots[s] = std::uint32_t(values_.size() + 1);
386 values_.push_back(v);
387 hashes_.push_back(h);
388 return std::uint32_t(values_.size() - 1);
389 }
390 if (hashes_[at - 1] == h && std::memcmp(&values_[at - 1], &v, sizeof(T)) == 0) return at - 1;
391 }
392 }
395 std::span<const T> values() const { return values_; }
396 std::size_t size() const { return values_.size(); }
397 void clear() {
398 values_.clear();
399 hashes_.clear();
400 std::fill(t_.slots.begin(), t_.slots.end(), 0u);
401 }
402
403 private:
404 std::vector<T> values_;
405 std::vector<std::uint64_t> hashes_;
406 enc_detail::probe_slots t_;
407};

457template <class T>
458struct value_stats {
459 bool has_minmax = false;
460 T min{}, max{};
461 std::int64_t null_count = 0;
462};
463template <class T>
464 requires(std::is_arithmetic_v<T>)
465inline value_stats<T> compute_stats(std::span<const T> values, const std::uint8_t* validity, std::size_t n) {
466 value_stats<T> s;
467 for (std::size_t i = 0; i < n; ++i) {
468 if (validity && !get_bit(validity, i)) {
469 ++s.null_count;
470 continue;
471 }
472 const T v = values[i];
473 if constexpr (std::is_floating_point_v<T>)
474 if (v != v) continue; // NaN
475 if (!s.has_minmax) {
476 s.min = s.max = v;
477 s.has_minmax = true;
478 } else {
479 if (v < s.min) s.min = v;
480 if (v > s.max) s.max = v;
481 }
482 }
483 if constexpr (std::is_floating_point_v<T>) {
484 if (s.has_minmax && s.min == T(0)) s.min = -T(0);
485 if (s.has_minmax && s.max == T(0)) s.max = T(0);
486 }
487 return s;
488}

492struct binary_stats {
493 bool has_minmax = false;
494 std::string_view min, max;
495};
496inline binary_stats compute_binary_stats(std::span<const std::string_view> views) {
497 binary_stats s;
498 const auto less = [](std::string_view a, std::string_view b) {
499 const std::size_t n = std::min(a.size(), b.size());
500 const int c = n ? std::memcmp(a.data(), b.data(), n) : 0; // memcmp compares as unsigned char
501 return c < 0 || (c == 0 && a.size() < b.size());
502 };
503 for (const auto v : views) {
504 if (!s.has_minmax) {
505 s.min = s.max = v;
506 s.has_minmax = true;
507 } else {
508 if (less(v, s.min)) s.min = v;
509 if (less(s.max, v)) s.max = v;
510 }
511 }
512 return s;
513}
