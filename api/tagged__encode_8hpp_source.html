2// nanom/tagged_encode.hpp — WRITE the tagged messages of nanom/tagged.hpp (Thrift compact).
3//
4// The same described struct that thrift_compact<M>() decodes is encoded here, into any byte sink:
5//
6// std::vector<std::byte> out;
7// auto n = nm::thrift_compact_encode(meta, out); // append; n = bytes written or an error
8//
9// std::array<std::byte, 256> buf; // fixed buffer: never written past
10// nm::span_sink s{buf};
11// if (auto r = nm::thrift_compact_encode(header, s); !r) ... // r.error(): what + where
12//
13// const std::size_t size = *nm::thrift_compact_size(meta); // exact, nothing written
14//
15// Building a message to write is zero-copy: std::string_view / nm::bytes members are views,
16// repeated members are nm::list<E>::of(span) over the caller's elements, nested messages
17// nm::lazy<M>::of(m). A list or message decoded from a file is written back verbatim (its wire
18// bytes are already valid), so read-modify-write keeps whatever the model does not declare.
19//
20// Writing is checked like reading:
21// * a string, binary or list longer than the wire's i32 is an error (never a truncated length);
22// * a lazy<M> that was neither decoded nor set with of() is an error (it has no bytes to write);
23// * a sink that refuses bytes (a full span_sink, a failed stream) is an error, and nothing is
24// written past a fixed buffer;
25// * field ids stay ascending and unique — the same static_asserts as decoding.
26// The error names the struct and field it happened in.
27//
28// Reading code never needs this header (nanom/tagged.hpp does not include it).

45template <byte_sink S>
46struct writer {
47 explicit writer(S& s) : sink(s) {}
48 S& sink;
49 std::size_t total = 0;
50 const char* err = nullptr;
51 std::string_view message, field;
52 std::size_t len = 0;
53 std::byte buf[256];
54
55 void fail(const char* what) {
56 if (!err) err = what;
57 }
58 void flush() {
59 if (err || !len) return;
60 if (!sink.put(buf, len)) return fail("the sink refused the bytes (full or failed)");
61 total += len;
62 len = 0;
63 }
64 void u8(std::uint8_t b) {
65 if (len == sizeof(buf)) flush();
66 if (err) return;
67 buf[len++] = std::byte(b);
68 }
69 void varint(std::uint64_t v) {
70 if (sizeof(buf) - len < leb128_max_bytes<std::uint64_t>) flush();
71 if (err) return;
72 len += uleb128_encode(v, buf + len);
73 }
74 void raw(const std::byte* p, std::size_t n) {
75 if (err || !n) return;
76 if (n <= sizeof(buf) - len) {
77 std::memcpy(buf + len, p, n);
78 len += n;
79 return;
80 }
81 flush();
82 if (err) return;
83 if (!sink.put(p, n)) return fail("the sink refused the bytes (full or failed)");
84 total += n;
85 }
87 bool length(std::size_t n) {
88 if (n > std::size_t(INT32_MAX)) {
89 fail("a string, binary or list longer than the wire's i32 length");
90 return false;
91 }
92 varint(n);
93 return true;
94 }
95 void list_header(std::size_t n, std::uint8_t et) {
96 if (n > std::size_t(INT32_MAX)) return fail("a list longer than the wire's i32 count");
97 if (n < 15) {
98 u8(std::uint8_t((n << 4) | et));
99 } else {
100 u8(std::uint8_t(0xf0 | et));
101 varint(n);
102 }
103 }
104};

108template <class T, class W>
109void write_value(W& w, const T& v) {
110 if constexpr (std::is_same_v<T, bool>) {
111 w.u8(v ? 1 : 2); // in a list; a bool field carries its value in the field header
112 } else if constexpr (std::is_same_v<T, std::int8_t>) {
113 w.u8(std::bit_cast<std::uint8_t>(v));
114 } else if constexpr (std::is_same_v<T, std::int16_t> || std::is_same_v<T, std::int32_t> ||
115 std::is_same_v<T, std::int64_t>) {
116 w.varint(zigzag_encode(v));
117 } else if constexpr (std::is_same_v<T, double>) {
118 const auto u = std::bit_cast<std::uint64_t>(v);
119 std::byte le[8];
120 for (int i = 0; i < 8; ++i) le[i] = std::byte(std::uint8_t(u >> (8 * i)));
121 w.raw(le, 8);
122 } else if constexpr (std::is_enum_v<T>) {
123 w.varint(zigzag_encode(std::int32_t(v)));
124 } else if constexpr (std::is_same_v<T, std::string_view> || std::is_same_v<T, std::string>) {
125 if (w.length(v.size())) w.raw(reinterpret_cast<const std::byte*>(v.data()), v.size());
126 } else if constexpr (std::is_same_v<T, bytes>) {
127 if (w.length(v.size())) w.raw(v.data(), v.size());
128 } else if constexpr (is_vector_t<T>::value) {
129 using E = typename T::value_type;
130 w.list_header(v.size(), type_of<E>());
131 for (const auto& e : v) write_value<E>(w, e);
132 } else if constexpr (is_list_t<T>::value) {
133 using E = typename T::value_type;
134 if (v.is_source()) {
135 const auto src = v.source();
136 w.list_header(src.size(), type_of<E>());
137 for (const auto& e : src) write_value<E>(w, e);
138 } else {
139 // decoded from the wire: the region is already-valid encoding of exactly these elements
140 w.list_header(v.size(), v.size() ? v.elem_wire() : type_of<E>());
141 w.raw(v.region().first, v.region().size());
142 }
143 } else if constexpr (is_lazy_t<T>::value) {
144 if (const auto* src = v.source()) write_struct(w, *src);
145 else if (!v.region().empty()) w.raw(v.region().first, v.region().size());
146 else w.fail("a nested message (lazy<M>) that was neither decoded nor set with lazy<M>::of()");
147 } else if constexpr (std::is_same_v<T, empty_struct>) {
148 w.u8(0);
149 } else if constexpr (Message<T>) {
150 write_struct(w, v);
151 } else {
152 static_assert(always_false<T>, "nanom thrift: unsupported member type");
153 }
154}

156template <Message M, class W>
157void write_struct(W& w, const M& m) {
158 static_assert(!message_info<M>::has_unknown, "nanom thrift: pb_unknown is protobuf-only");
159 const auto saved_message = w.message, saved_field = w.field;
160 w.message = describe<M>::name();
161 std::int32_t last = 0;
162 for_each_field<M>([&](auto f) {
163 if (w.err) return;
164 using F = member_t<decltype(f)::mem_ptr>;
165 using V = typename F::value_type;
166 w.field = decltype(f)::name.sv();
167 const V& fv = (m.*(decltype(f)::mem_ptr)).v;
168 const auto emit = [&](const auto& val) {
169 using T = std::remove_cvref_t<decltype(val)>;
170 std::uint8_t type = type_of<T>();
171 if constexpr (std::is_same_v<T, bool>)
172 type = val ? std::uint8_t(ctype::bool_true) : std::uint8_t(ctype::bool_false);
173 const std::int32_t delta = std::int32_t(F::id) - last;
174 if (delta >= 1 && delta <= 15) {
175 w.u8(std::uint8_t((delta << 4) | type));
176 } else {
177 w.u8(type);
178 w.varint(zigzag_encode(std::int16_t(F::id)));
179 }
180 last = F::id;
181 if constexpr (!std::is_same_v<T, bool>) write_value<T>(w, val);
182 };
183 if constexpr (is_optional_t<V>::value) {
184 if (fv) emit(*fv);
185 } else {
186 emit(fv);
187 }
188 });
189 if (w.err) return; // keep the innermost struct / field as the error location
190 w.u8(0);
191 w.message = saved_message;
192 w.field = saved_field;
193}

199template <Message M, byte_sink S>
200expected<std::size_t, encode_error> thrift_compact_encode(const M& m, S& sink) {
201 detail::tc::writer<S> w(sink);
202 detail::tc::write_struct(w, m);
203 w.flush();
204 if (w.err) return unexpected<encode_error>(encode_error{w.err, w.message, w.field});
205 return w.total;
206}
