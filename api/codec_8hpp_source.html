2// nanom/codec.hpp — page decompression with no external dependencies: Snappy (raw) and LZ4 block.
3//
4// Columnar formats compress each page independently and record the uncompressed size in the page
5// header. These decoders take that size as the output span and never write past it, never read
6// past the input, and reject every back-reference that points before the start of the output — a
7// corrupt or hostile page fails with a codec_error, it never becomes an out-of-bounds access. The
8// "decompression bomb" defence is the caller's: size `out` from the header's uncompressed size
9// after capping it (the decoders cannot produce more than out.size() bytes).
10//
11// Heavier codecs (zstd, gzip/deflate, brotli) plug in at the reader through the same shape:
12// expected<std::size_t, codec_error> decompress(std::span<const std::byte> in, std::span<std::byte> out)

30inline void copy_match(std::byte* dst, std::size_t offset, std::size_t len) {
31 const std::byte* src = dst - offset;
32 if (offset >= len) {
33 std::memcpy(dst, src, len);
34 } else if (offset >= 8) {
35 while (len >= 8) {
36 std::memcpy(dst, src, 8);
37 dst += 8; src += 8; len -= 8;
38 }
39 while (len--) *dst++ = *src++;
40 } else {
41 while (len--) *dst++ = *src++;
42 }
43}
44inline void copy8(std::byte* d, const std::byte* s) { std::memcpy(d, s, 8); }
45inline void copy16(std::byte* d, const std::byte* s) { std::memcpy(d, s, 16); }
46
51inline void copy_match_wild(std::byte* dst, std::size_t offset, std::size_t len) {
52 const std::byte* src = dst - offset;
53 if (offset >= 16) {
54 for (std::size_t i = 0; i < len; i += 16) copy16(dst + i, src + i);
55 } else if (offset >= 8) {
56 for (std::size_t i = 0; i < len; i += 8) copy8(dst + i, src + i);
57 } else {
58 // offset 1..7 (run-like repeats): the first 8 bytes byte-wise, then 8-byte chunks from a lag
59 // that is a multiple of the period and >= 8 (same bytes, by periodicity, and non-overlapping)
60 const std::size_t head = len < 8 ? len : 8;
61 for (std::size_t i = 0; i < head; ++i) dst[i] = src[i];
62 const std::size_t lag = offset * ((8 + offset - 1) / offset); // 8..14
63 for (std::size_t i = head; i < len; i += 8) copy8(dst + i, dst + i - lag);
64 }
65}

69// 30. Snappy (raw block format — what Parquet's SNAPPY codec stores)
70// ---------------------------------------------------------------------------
71//
72// Fast paths (all guarded by explicit room checks; the exact paths below handle the last bytes of
73// a block and every other case): short literals move as one fixed 16-byte copy, back-references
74// as 16- / 8-byte chunks or periodic 8-byte chunks for offsets under 8. Every read stays inside
75// the input or the already-produced output; every write stays inside `out`.
76
78inline std::optional<std::uint32_t> snappy_uncompressed_length(std::span<const std::byte> in) {
79 auto r = varint_u32(from(in));
80 if (!r) return std::nullopt;
81 return r->value;
82}

84namespace detail {
88consteval std::array<std::uint16_t, 256> snappy_table() {
89 std::array<std::uint16_t, 256> t{};
90 for (unsigned tag = 0; tag < 256; ++tag) {
91 unsigned len = 0, high = 0, extra = 0;
92 switch (tag & 3) {
93 case 0: len = (tag >> 2) + 1; if (len > 60) len = 0; extra = 0; break;
94 case 1: len = ((tag >> 2) & 7) + 4; high = (tag >> 5) << 8; extra = 1; break;
95 case 2: len = (tag >> 2) + 1; extra = 2; break;
96 default: len = (tag >> 2) + 1; extra = 4; break;
97 }
98 t[tag] = std::uint16_t(len | high | (extra << 11));
99 }
100 return t;
101}
102inline constexpr std::array<std::uint16_t, 256> snappy_tags = snappy_table();
103inline constexpr std::uint32_t snappy_extra_mask[5] = {0, 0xff, 0xffff, 0xffffff, 0xffffffff};
104inline std::uint32_t load_le32(const std::byte* p) {
105 std::uint32_t v;
106 std::memcpy(&v, p, 4);
107 if constexpr (std::endian::native == std::endian::big) v = std::byteswap(v);
108 return v;
109}
110} // namespace detail

113inline status snappy_decompress(std::span<const std::byte> in, std::span<std::byte> out) {
114 auto pre = varint_u32(from(in));
115 if (!pre) return detail::fail("snappy: bad length preamble", 0);
116 if (pre->value != out.size()) return detail::fail("snappy: length preamble does not match the page header", 0);
117 const std::byte* ip = pre->rest.first;
118 const std::byte* const ie = in.data() + in.size();
119 std::byte* op = out.data();
120 std::byte* const ob = out.data();
121 std::byte* const oe = out.data() + out.size();
122 const auto at = [&] { return std::size_t(ip - in.data()); };
123 const auto u8 = [](const std::byte* p) { return std::size_t(std::uint8_t(*p)); };
124 // fast-path limits, computed once (an element may use the fast path while ip < ip_fast and
125 // op < op_fast: 16 input bytes and 80 output bytes of headroom)
126 const std::byte* const ip_fast = in.size() >= 16 ? ie - 16 : in.data();
127 std::byte* const op_fast = out.size() >= 80 ? oe - 80 : ob;
128 while (ip < ie) {
129 const std::size_t tag = u8(ip);
130 const std::uint16_t entry = detail::snappy_tags[tag];
131 // Fast path, per element: while >= 16 input bytes (tag + 4 extra bytes, or a 16-byte literal)
132 // and >= 80 output bytes (the longest copy, 64, plus 16 of overshoot) remain, an element is a
133 // table lookup, one masked 4-byte load and fixed-size copies. Offsets are still validated.
134 if (ip < ip_fast && op < op_fast) {
135 // Literals and copies take ONE branch-free step: the source is selected (cmov), not
136 // branched on, so the literal/copy alternation of typical data costs no mispredictions.
137 // It covers literals of 1..16 bytes and copies of <= 16 bytes from >= 8 back, as two
138 // 8-byte moves (with offset >= 8 the second move reads only bytes already final).
139 const bool lit = (tag & 3) == 0;
140 const std::size_t len = entry & 0xff; // 0: a literal with a length suffix
141 const std::size_t extra = entry >> 11;
142 const std::size_t offset = (entry & 0x700) + (detail::load_le32(ip + 1) & detail::snappy_extra_mask[extra]);
143 const std::size_t produced = std::size_t(op - ob);
144 // the next tag's position comes from the tag by ALU alone (no table load on the loop's
145 // ip -> tag -> ip dependency chain): a literal advances (tag >> 2) + 2, a copy-1 / copy-2
146 // (tag & 3) + 1; copy-4 (tag & 3 == 3) takes the path below
147 const bool slow = (len - 1 >= 16) | ((tag & 3) == 3) | (bool(!lit) & ((offset < 8) | (offset > produced)));
148 if (!slow) {
149 const std::byte* src = lit ? ip + 1 : op - offset;
150 detail::copy8(op, src);
151 detail::copy8(op + 8, src + 8);
152 op += len;
153 ip += lit ? (tag >> 2) + 2 : (tag & 3) + 1;
154 continue;
155 }
156 if (!lit && len <= 64 && offset - 1 < produced) {
157 // a copy the step above does not cover: longer than 16 or from under 8 back
158 ip += 1 + extra;
159 detail::copy_match_wild(op, offset, len);
160 op += len;
161 continue;
162 }
163 // a literal over 16 bytes, or an invalid offset: the exact path below
164 }
165 ++ip; // exact path: every length and offset checked against both ends
166 if ((tag & 3) == 0) { // literal
167 std::size_t len = (tag >> 2) + 1;
168 if (len > 60) {
169 const std::size_t nb = len - 60; // 1..4 length bytes
170 if (std::size_t(ie - ip) < nb) return detail::fail("snappy: truncated literal length", at());
171 len = 0;
172 for (std::size_t i = 0; i < nb; ++i) len |= u8(ip + i) << (8 * i);
173 ip += nb;
174 len += 1;
175 }
176 if (std::size_t(ie - ip) < len) return detail::fail("snappy: literal runs past the input", at());
177 if (std::size_t(oe - op) < len) return detail::fail("snappy: literal overflows the output", at());
178 std::memcpy(op, ip, len);
179 ip += len;
180 op += len;
181 continue;
182 }
183 std::size_t len, offset;
184 switch (tag & 3) {
185 case 1:
186 if (ie - ip < 1) return detail::fail("snappy: truncated copy", at());
187 len = ((tag >> 2) & 7) + 4;
188 offset = ((tag >> 5) << 8) | u8(ip);
189 ip += 1;
190 break;
191 case 2:
192 if (ie - ip < 2) return detail::fail("snappy: truncated copy", at());
193 len = (tag >> 2) + 1;
194 offset = u8(ip) | (u8(ip + 1) << 8);
195 ip += 2;
196 break;
197 default:
198 if (ie - ip < 4) return detail::fail("snappy: truncated copy", at());
199 len = (tag >> 2) + 1;
200 offset = u8(ip) | (u8(ip + 1) << 8) | (u8(ip + 2) << 16) | (u8(ip + 3) << 24);
201 ip += 4;
202 break;
203 }
204 if (offset == 0 || offset > std::size_t(op - ob)) return detail::fail("snappy: copy offset outside the output", at());
205 if (std::size_t(oe - op) >= len + 16) {
206 detail::copy_match_wild(op, offset, len);
207 } else {
208 if (std::size_t(oe - op) < len) return detail::fail("snappy: copy overflows the output", at());
209 detail::copy_match(op, offset, len);
210 }
211 op += len;
212 }
213 if (op != oe) return detail::fail("snappy: output shorter than declared", at());
214 return out.size();
215}

223inline status lz4_block_decompress(std::span<const std::byte> in, std::span<std::byte> out) {
224 const std::byte* ip = in.data();
225 const std::byte* const ie = ip + in.size();
226 std::byte* op = out.data();
227 std::byte* const ob = out.data();
228 std::byte* const oe = out.data() + out.size();
229 const auto at = [&] { return std::size_t(ip - in.data()); };
230 const auto ext_len = [&](std::size_t& len) {
231 if (len != 15) return true;
232 for (;;) {
233 if (ip >= ie) return false;
234 const std::size_t b = std::size_t(std::uint8_t(*ip++));
235 if (len > std::numeric_limits<std::size_t>::max() - b) return false;
236 len += b;
237 if (b != 255) return true;
238 }
239 };
240 if (in.empty()) return detail::fail("lz4: empty block", 0);
241 for (;;) {
242 if (ip >= ie) return detail::fail("lz4: truncated sequence", at());
243 const std::size_t token = std::size_t(std::uint8_t(*ip++));
244 std::size_t lit = token >> 4;
245 if (lit < 15 && ie - ip >= 16 && oe - op >= 16) {
246 detail::copy16(op, ip);
247 ip += lit;
248 op += lit;
249 } else {
250 if (!ext_len(lit)) return detail::fail("lz4: truncated literal length", at());
251 if (std::size_t(ie - ip) < lit) return detail::fail("lz4: literals run past the input", at());
252 if (std::size_t(oe - op) < lit) return detail::fail("lz4: literals overflow the output", at());
253 if (lit) std::memcpy(op, ip, lit); // lit may be 0 into an empty output: no null memcpy
254 ip += lit;
255 op += lit;
256 }
257 if (ip == ie) break; // the last sequence carries literals only
258 if (ie - ip < 2) return detail::fail("lz4: truncated match offset", at());
259 const std::size_t offset = std::size_t(std::uint8_t(ip[0])) | (std::size_t(std::uint8_t(ip[1])) << 8);
260 ip += 2;
261 if (offset == 0 || offset > std::size_t(op - ob)) return detail::fail("lz4: match offset outside the output", at());
262 std::size_t mlen = token & 15;
263 if (!ext_len(mlen)) return detail::fail("lz4: truncated match length", at());
264 mlen += 4;
265 if (std::size_t(oe - op) >= mlen + 16) {
266 detail::copy_match_wild(op, offset, mlen);
267 } else {
268 if (std::size_t(oe - op) < mlen) return detail::fail("lz4: match overflows the output", at());
269 detail::copy_match(op, offset, mlen);
270 }
271 op += mlen;
272 }
273 return std::size_t(op - ob);
274}
