18inline std::int64_t int96_to_unix_nanos(const std::byte* v) {
19 std::uint64_t nanos;
20 std::uint32_t jd;
21 std::memcpy(&nanos, v, 8);
22 std::memcpy(&jd, v + 8, 4);
23 if constexpr (std::endian::native == std::endian::big) {
24 nanos = std::byteswap(nanos);
25 jd = std::byteswap(jd);
26 }
27 return std::int64_t(std::uint64_t(std::int64_t(jd) - 2440588) * 86400000000000ull + nanos);
28}

32inline col::kernel_status dictionary_indices(std::span<const std::byte> vals, std::size_t nn, std::size_t dict_size,
33 std::span<std::uint32_t> out) {
34 if (out.size() < nn) return col::kernel_fail("output smaller than the value count");
35 if (nn == 0) return col::kernel_ok;
36 if (vals.empty()) return col::kernel_fail("dictionary indices missing");
37 const unsigned bw = std::uint8_t(vals[0]);
38 if (bw > 32) return col::kernel_fail("dictionary index bit width over 32");
39 col::rle_bp_decoder d(vals.subspan(1), bw);
40 if (d.get(out.data(), nn) != nn || !d.ok()) return col::kernel_fail("dictionary indices shorter than the page");
41 if (!col::check_indices(out.first(nn), dict_size)) return col::kernel_fail("dictionary index out of range");
42 return col::kernel_ok;
43}
