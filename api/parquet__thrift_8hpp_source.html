1// SPDX-License-Identifier: Apache-2.0
2// nanom/formats/parquet_thrift.hpp — the Parquet file metadata (parquet.thrift) as reflected
3// nanom tagged messages, plus the footer locator.
4//
5// This is the shared model the Parquet reader (parquet2nanoarrow) and writer (nanoarrow2parquet)
6// both build on: one declaration per Thrift struct, decoded and encoded by nanom/tagged.hpp's
7// Thrift compact codec. Only the fields a reader needs are declared; everything else (encryption,
8// geospatial and size statistics, …) is skipped structurally, so newer writers stay readable.
9//
10// Field ids and types follow apache/parquet-format's parquet.thrift. Enums are OPEN (int32-backed):
11// an enumerator a newer writer added round-trips as its number instead of failing the decode.
12//
13// The model lives in nanom_formats::parquet, not under nanom::, on purpose: C++26 reflection
14// auto-describes only types outside the library's own namespace (see nanom26.hpp), so these structs
15// need no registration there; under C++23 the NANOM_DESCRIBE lines at the bottom register them.
16//
17// Zero-copy: strings and binaries are views into the footer buffer; repeated members are lazy
18// nm::list<> views, so e.g. `meta.row_groups->at(7)` decodes one row group out of thousands without
19// touching the rest. The footer buffer must outlive every view taken from it.

287inline constexpr std::string_view magic = "PAR1";
288inline constexpr std::string_view encrypted_footer_magic = "PARE";
289
291struct footer_location {
292 std::uint64_t metadata_offset;
293 std::uint32_t metadata_length;
294};
295
299inline result<footer_location> locate_footer(input tail8, std::uint64_t file_size) {
300 if (tail8.size() < 8) return make_incomplete(tail8, 8 - tail8.size());
301 const input t = tail8.advance(tail8.size() - 8);
302 auto len = le_u32(t);
303 if (!len) return unexp(len.error());
304 const std::string_view m = as_str(len->rest.take_span(4));
305 if (m == encrypted_footer_magic)
306 return make_err(t, "a plaintext Parquet footer (encrypted footers are not supported)");
307 if (m != magic) return make_err(len->rest, "Parquet magic \"PAR1\"");
308 if (file_size < 12 || len->value > file_size - 12)
309 return make_err(t, "a Parquet footer length that fits inside the file");
310 return done{footer_location{file_size - 8 - len->value, len->value}, t.advance(8)};
311}
312
315inline result<FileMetaData> read_file_metadata(input file) {
316 auto loc = locate_footer(file, file.size());
317 if (!loc) return unexp(loc.error());
318 if (file.size() < 4 || as_str(file.take_span(4)) != magic)
319 return make_err(file, "leading Parquet magic \"PAR1\"");
320 const input meta = file.advance(std::size_t(loc->value.metadata_offset));
321 const input exact = meta.with_range(meta.first, meta.first + loc->value.metadata_length);
322 auto r = thrift_compact<FileMetaData>()(exact);
323 if (!r) return r;
324 if (!r->rest.empty()) return make_err(r->rest, "FileMetaData to span the whole footer");
325 return done{std::move(r->value), meta.advance(meta.size())};
326}
