2// nanom/segmented.hpp — parsing over DISJOINT byte ranges (scatter-gather / "strudel" input).
3//
4// The core cursor (nom.hpp's `input`) is a contiguous [first,last) pointer pair, and stays that
5// way: this header is the opt-in sibling for the one case that model cannot express — a logical
6// buffer whose bytes live in several disjoint spans (the motivating consumer is IP fragment
7// reassembly, examples/nano_shark/core/defrag.hpp, where a reassembled datagram's fragments sit
8// scattered through the capture file). Design contract, in order of priority:
9//
10// 1. ZERO COST WHEN UNUSED. Nothing in nom.hpp/reflect.hpp changes; `input`, its layout, and
11// every combinator are untouched. Don't include this header, don't pay for it.
12// 2. PAY ONLY AT THE SEAMS. A read that lies inside one segment compiles to the same
13// pointer-based code as the contiguous cursor; only a read that STRADDLES a segment
14// boundary pays — a bounded, stack-only gather of at most one struct's wire_size_v<T>
15// bytes (seg_window's inline buffer). Never a heap allocation, never a whole-buffer copy.
16// 3. HONEST ZERO-COPY VIEWS. overlay_seg<T> hands out view<T> pointers into caller-owned
17// segment memory or fails with a recoverable error — it never points a view at a hidden
18// temporary. Straddling callers use strct_seg<T> (by value) instead.
19//
20// Deliberately NOT supported over segments (v1): the general combinator vocabulary
21// (alt/many0/tag/take_until/dec/hex/...) — the text-oriented members need physically contiguous
22// memory (`from_chars`, `memcmp`, `std::search`) and cannot be segmented without copies. What IS
23// here: the struct parsers (strct_seg/overlay_seg), a small cursor kit for hand-rolled walkers
24// (seg_u8/seg_be16/seg_be32/...), and zero-copy narrowing (subrange). Bit FIELDS inside a
25// described struct work fine — they decode from the gathered window like every other field.

71 [[nodiscard]] constexpr std::uint8_t byte_at(std::size_t i) const {
72 for (const auto& p : parts_) {

90#endif

128struct seg_input {
129 segments src{};
130 std::size_t seg_i = 0;
131 const std::byte* cur = nullptr;
132 const std::byte* seg_end = nullptr;
133 std::size_t abs = 0;
134 bool live = false;
135
136 constexpr seg_input() = default;
137 explicit constexpr seg_input(segments s) : src(s) { enter(0); }
138
139 [[nodiscard]] constexpr std::size_t size() const { return src.size() - abs; }
140 [[nodiscard]] constexpr bool empty() const { return cur == nullptr; }
141 [[nodiscard]] constexpr std::size_t offset() const { return abs; }
142
144 [[nodiscard]] constexpr std::span<const std::byte> contiguous() const {
145 return cur ? std::span<const std::byte>(cur, std::size_t(seg_end - cur))
146 : std::span<const std::byte>{};
147 }
148
150 [[nodiscard]] constexpr std::uint8_t operator[](std::size_t i) const {
151 const std::byte* c = cur;
152 const std::byte* e = seg_end;
153 std::size_t si = seg_i;
154 while (true) {
155 const std::size_t here = std::size_t(e - c);
156 if (i < here) return std::uint8_t(c[i]);
157 i -= here;
158 do { ++si; } while (src.part(si).empty()); // precondition guarantees a next byte exists
159 const auto p = src.part(si);
160 c = p.data();
161 e = p.data() + p.size();
162 }
163 }
164
166 [[nodiscard]] constexpr std::optional<std::uint8_t> safe_at(std::size_t i) const {
167 if (i >= size()) return std::nullopt;
168 return (*this)[i];
169 }
170
173 [[nodiscard]] constexpr seg_input advance(std::size_t n) const {
174 seg_input o = *this;
175 o.abs += n;
176 while (o.cur) {
177 const std::size_t here = std::size_t(o.seg_end - o.cur);
178 if (n < here) {
179 o.cur += n;
180 return o;
181 }
182 n -= here;
183 o.enter(o.seg_i + 1);
184 if (n == 0) return o;
185 }
186 o.abs = src.size(); // clamped
187 return o;
188 }
189
191 [[nodiscard]] constexpr std::optional<seg_input> checked_advance(std::size_t n) const {
192 if (n > size()) return std::nullopt;
193 return advance(n);
194 }

199 template <std::size_t N>
200 constexpr bool gather(seg_window<N>& w) const {
201 if (cur != nullptr && std::size_t(seg_end - cur) >= N) { // fast path first
202 w.ext_ = cur;
203 return true;
204 }
205 if (size() < N) return false;
206 w.ext_ = nullptr;
207 std::size_t got = 0;
208 const std::byte* c = cur;
209 const std::byte* e = seg_end;
210 std::size_t si = seg_i;
211 while (got < N) {
212 const std::size_t here = std::min(N - got, std::size_t(e - c));
213 for (std::size_t k = 0; k < here; ++k) w.buf_[got + k] = c[k]; // -O2 lowers to memcpy
214 got += here;
215 if (got == N) break;
216 do { ++si; } while (src.part(si).empty()); // size() >= N guarantees a next part
217 const auto p = src.part(si);
218 c = p.data();
219 e = p.data() + p.size();
220 }
221 return true;
222 }
223
225 template <std::size_t N>
226 [[nodiscard]] constexpr std::optional<seg_window<N>> gather() const {
227 std::optional<seg_window<N>> out(std::in_place);
228 if (!gather(*out)) return std::nullopt;
229 return out;
230 }
231
233 constexpr bool gather(std::span<std::byte> dst) const {
234 if (size() < dst.size()) return false;
235 std::size_t got = 0;
236 const std::byte* c = cur;
237 const std::byte* e = seg_end;
238 std::size_t si = seg_i;
239 while (got < dst.size()) {
240 const std::size_t here = std::min(dst.size() - got, std::size_t(e - c));
241 for (std::size_t k = 0; k < here; ++k) dst[got + k] = c[k];
242 got += here;
243 if (got == dst.size()) break;
244 do { ++si; } while (src.part(si).empty());
245 const auto p = src.part(si);
246 c = p.data();
247 e = p.data() + p.size();
248 }
249 return true;
250 }
251
254 [[nodiscard]] constexpr std::optional<seg_subrange> subrange(std::size_t n) const;
255
256#if NANOM_GENERATION
257 [[nodiscard]] constexpr const wire_arena* arena() const { return src.arena(); }
258 [[nodiscard]] constexpr std::uint64_t gen() const { return src.gen(); }
259#endif
260
261 private:
263 constexpr void enter(std::size_t i) {
264 seg_i = i;
265 while (seg_i < src.parts()) {
266 const auto p = src.part(seg_i);
267 if (!p.empty()) {
268 cur = p.data();
269 seg_end = p.data() + p.size();
270 return;
271 }
272 ++seg_i;
273 }
274 cur = seg_end = nullptr;
275 }
276};

350constexpr std::optional<seg_subrange> seg_input::subrange(std::size_t n) const {
351 if (n > size()) return std::nullopt;
352 seg_subrange out;
353#if NANOM_GENERATION
354 out.arena_ = arena();
355 out.gen_ = gen();
356#endif
357 out.total_ = n;
358 const std::byte* c = cur;
359 const std::byte* e = seg_end;
360 std::size_t si = seg_i;
361 while (n > 0) {
362 if (out.count_ == seg_max_parts) return std::nullopt; // over the inline cap
363 const std::size_t here = std::min(n, std::size_t(e - c));
364 out.parts_[out.count_++] = std::span<const std::byte>(c, here);
365 n -= here;
366 if (n == 0) break;
367 do { ++si; } while (src.part(si).empty());
368 const auto p = src.part(si);
369 c = p.data();
370 e = p.data() + p.size();
371 }
372 return out;
373}

391inline unexpected<error> seg_make_err(const seg_input& at, const char* expected) {
392 error e;
393 e.kind = errk::err;
394 e.offset = std::uint32_t(at.offset());
395 e.expected = expected;
396 return unexp(e);
397}
398inline unexpected<error> seg_make_incomplete(const seg_input& at, std::size_t needed) {
399 error e;
400 e.kind = at.live ? errk::incomplete : errk::err;
401 e.offset = std::uint32_t(at.offset() + at.size());
402 e.expected = "more input";
403 e.needed = std::uint32_t(std::min(needed, std::size_t(max_incomplete_needed)));
404 return unexp(e);
405}

408inline std::string render(const error& e, const segments& whole) {
409 std::string out;
410 out += "parse ";
411 out += e.kind == errk::incomplete ? "incomplete" : e.kind == errk::fail ? "failure" : "error";
412 out += " at offset " + std::to_string(e.offset);
413 if (e.needed) out += " (need " + std::to_string(e.needed) + " more byte(s))";
414 out += ": expected ";
415 out += e.expected;
416 for (std::uint8_t i = 0; i < e.nctx; ++i) {
417 out += "\n in ";
418 out += e.ctx[i].label;
419 out += " (starting at offset " + std::to_string(e.ctx[i].offset) + ")";
420 }
421 const std::size_t total = whole.size();
422 const std::size_t off = std::min<std::size_t>(e.offset, total);
423 if (e.offset > total) out += " (offset beyond input, hex window clamped)";
424 if (total > 0) {
425 const std::size_t lo = off >= 8 ? off - 8 : 0;
426 const std::size_t hi = std::min(off + 8, total);
427 static constexpr char hexd[] = "0123456789abcdef";
428 std::string line = " ", caret = " ";
429 for (std::size_t i = lo; i < hi; ++i) {
430 const auto b = whole.byte_at(i);
431 line += hexd[b >> 4];
432 line += hexd[b & 15];
433 line += ' ';
434 caret += (i == off) ? "^^ " : " ";
435 }
436 if (off == total) caret += "^^ (end of input)";
437 out += "\n" + line + "\n" + caret;
438 }
439 return out;
440}

443// struct parsing over segments — strct_seg<T>() / overlay_seg<T>()
444// ---------------------------------------------------------------------------

450template <Described T>
451constexpr auto strct_seg(std::endian dflt = std::endian::native) {
452 static_assert(detail::layout_ok<T>(),
453 "nanom: bit fields must pack to byte boundaries and every "
454 "non-bit field must start byte-aligned");
455 return [dflt](seg_input in) -> seg_result<T> {
456 constexpr std::size_t need = wire_size_v<T>;
457 seg_window<need> w;
458 if (!in.gather(w)) return seg_make_incomplete(in, need - in.size());
459 T out{};
460 constexpr auto offs = detail::field_bit_offsets<T>();
461 std::size_t i = 0;
462 detail::for_each_field<T>([&](auto f) {
463 out.*(decltype(f)::mem_ptr) =
464 detail::assign_field<detail::member_t<decltype(f)::mem_ptr>>(w.data(), offs[i++], dflt);
465 });
466 return seg_done{std::move(out), in.advance(need)};
467 };
468}

470// ---------------------------------------------------------------------------
471// cursor kit — scalar reads for hand-rolled walkers (SOME/IP-style TLV cursors)
472// ---------------------------------------------------------------------------
473// Each is a thin wrapper over gather<N>/advance, replacing `rd_be16(p + off)`-style raw pointer
474// arithmetic in consumer code. All return seg_result so short input reports exactly like the
475// struct parsers (incomplete on a live cursor, recoverable err otherwise).
476
477namespace detail_seg {
478template <std::size_t N, std::endian E>
479constexpr seg_result<detail::uint_for_bytes<N>> seg_uint(seg_input in) {
480 seg_window<N> w;
481 if (!in.gather(w)) return seg_make_incomplete(in, N - in.size());
482 using U = detail::uint_for_bytes<N>;
483 U u = 0;
484 const std::byte* p = w.data();
485 if constexpr (E == std::endian::big)
486 for (std::size_t i = 0; i < N; ++i) u = U(U(u << 8) | std::uint8_t(p[i]));
487 else
488 for (std::size_t i = 0; i < N; ++i) u |= U(U(std::uint8_t(p[i])) << (8 * i));
489 return seg_done{u, in.advance(N)};
490}
491} // namespace detail_seg

532template <Described T>
533constexpr auto overlay_seg(std::endian dflt = std::endian::native) {
534 static_assert(detail::layout_ok<T>(),
535 "nanom: bit fields must pack to byte boundaries and every "
536 "non-bit field must start byte-aligned");
537 return [dflt](seg_input in) -> seg_result<view<T>> {
538 constexpr std::size_t need = wire_size_v<T>;
539 if (in.size() < need) return seg_make_incomplete(in, need - in.size());
540 if (in.contiguous().size() < need)
541 return seg_make_err(in, "contiguous view (struct straddles a segment boundary; use strct_seg)");
542#if NANOM_GENERATION
543 return seg_done{view<T>{in.contiguous().data(), dflt, in.arena(), in.gen()}, in.advance(need)};
544#else
545 return seg_done{view<T>{in.contiguous().data(), dflt}, in.advance(need)};
546#endif
547 };
