2// nanom/formats/lance_encodings.hpp — Lance's page encodings (protobuf) as reflected nanom
3// messages: the format 2.1+ PageLayout / CompressiveEncoding tree (lance.encodings21) and the
4// format 2.0 ArrayEncoding tree (lance.encodings).
5//
6// Every data page of a Lance file carries one of these, wrapped in an Any-style message
7// ({1: type_url, 2: value}), in ColumnMetadata.Page.encoding (nanom/formats/lance_protobuf.hpp):
8//
9// auto any = nm::protobuf<lance::EncodingAny>()(page_encoding_bytes); // the wrapper
10// auto layout = nm::protobuf<lance::PageLayout>()(nm::from(*any->value.value));
11//
12// Both trees are recursive (an encoding wraps another), so every CompressiveEncoding / ArrayEncoding
13// a message refers to is an nm::pb_lazy<>: its bytes are checked to fit when the parent is read,
14// and decoded when a reader walks into them (child.decode_into(node)), which keeps a decode free of
15// allocations and the decoded structs small however deep the tree. A
16// reader walking the tree bounds its depth. A oneof is a set of std::optional members, at most one
17// of which a valid message sets. Scalars are
18// declared std::uint64_t so a reader decides how to narrow them (Lance's own readers cast).
19// Variants not declared here (new ones Lance adds) are kept in `unknown`, so a reader can refuse
20// them by field number instead of misreading the page.
21//
22// Field numbers follow lance-format/lance's protos/encodings_v2_1.proto and encodings_v2_0.proto,
23// as nanolance verified them against files written by Lance itself.

257template <class M>
258std::size_t oneof_members(const M& m) {
259 std::size_t n = 0;
260 detail::for_each_field<M>([&](auto f) {
261 using V = typename detail::member_t<decltype(f)::mem_ptr>::value_type;
262 const auto& v = (m.*(decltype(f)::mem_ptr)).v;
263 if constexpr (std::is_same_v<V, unknown_fields>) {
264 n += detail::pb::unknown_len_fields(v);
265 } else if constexpr (detail::is_optional_t<V>::value || detail::is_box_t<V>::value) {
266 n += v.has_value() ? 1 : 0;
267 }
268 });
269 return n;
270}
