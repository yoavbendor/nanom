1// SPDX-License-Identifier: Apache-2.0
2// nanom/fsst_encode.hpp — FSST compression: train a symbol table, compress values with it, and
3// serialize the table as Lance stores it. The inverse of nanom/fsst.hpp.
4//
5// Training follows the paper and Lance's build_symbol_table (rust/compression/fsst): six rounds over
6// a ~16 KiB sample, each compressing the sample with the current table, counting every symbol and
7// every adjacent pair, and keeping the 255 candidates with the highest gain (count x length, single
8// bytes x8); the table of the round that compressed the sample best wins. Three deliberate
9// differences, none visible to a decoder: the sample is drawn with a fixed seed, so the same input
10// always gives the same table; a candidate is identified by its bytes, so one string reached two
11// ways is one candidate with the summed gain; and matching never reads past the end of a value.
12//
13// Compression takes, at each position, the longest symbol that matches, else an escape and the
14// literal byte, so a value compresses to at most 2x its size.
15//
16// Symbols are matched against 8-byte words loaded in native order, as Lance's encoder does: the
17// encoder assumes a little-endian host. The table it serializes is little-endian either way.

65inline std::uint64_t hash3(std::uint64_t word) {
66 const auto w = word & 0xFFFFFF;
67 const auto m = w * 2971215073ULL; // FSST_HASH_PRIME
68 return (m ^ (m >> 15)) & (kLongSlots - 1);
69}

71inline std::uint64_t mask(unsigned length) {
72 return length >= 8 ? ~std::uint64_t{0} : ((std::uint64_t{1} << (8 * length)) - 1);
73}

76inline std::uint64_t load(const std::uint8_t* p, std::size_t remaining) {
77 std::uint64_t w = 0;
78 std::memcpy(&w, p, std::min<std::size_t>(remaining, 8));
79 return w;
80}

91inline bool add(encoder& e, std::uint64_t value, unsigned length) {
92 const auto code = static_cast<std::uint16_t>(e.symbol_count);
93 if (length == 1) {
94 e.byte_codes[value & 0xFF] = code;
95 } else if (length == 2) {
96 e.short_codes[value & 0xFFFF] = code;
97 } else {
98 auto& s = e.long_codes[hash3(value)];
99 if (s.length != 0) return false;
100 s = encoder::slot{value, static_cast<std::uint8_t>(length), static_cast<std::uint8_t>(code)};
101 }
102 e.symbols[code] = value;
103 e.lengths[code] = static_cast<std::uint8_t>(length);
104 ++e.symbol_count;
105 return true;
106}

109inline std::uint16_t find(const encoder& e, const std::uint8_t* p, std::size_t remaining, unsigned& length) {
110 const auto word = load(p, remaining);
111 if (remaining >= 3) {
112 const auto& s = e.long_codes[hash3(word)];
113 if (s.length != 0 && s.length <= remaining && (word & mask(s.length)) == s.value) {
114 length = s.length;
115 return s.code;
116 }
117 }
118 if (remaining >= 2) {
119 const auto code = e.short_codes[word & 0xFFFF];
120 if (code != encoder::kNone) {
121 length = 2;
122 return code;
123 }
124 }
125 length = 1;
126 return e.byte_codes[word & 0xFF];
127}

133template <class ValueAt>
134value_list make_sample(std::size_t n, ValueAt& value_at) {
135 std::size_t total = 0;
136 for (std::size_t i = 0; i < n; ++i) total += value_at(i).size();
137 value_list sample;
138 if (total <= kSampleTarget) {
139 sample.reserve(n);
140 for (std::size_t i = 0; i < n; ++i) sample.push_back(value_at(i));
141 return sample;
142 }
143 std::uint64_t state = 0x9E3779B97F4A7C15ULL ^ total;
144 std::size_t taken = 0;
145 while (taken < kSampleTarget) {
146 state += 0x9E3779B97F4A7C15ULL;
147 auto z = state;
148 z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
149 z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
150 z ^= z >> 31;
151 const std::span<const std::byte> v = value_at(static_cast<std::size_t>(z % n));
152 sample.push_back(v);
153 taken += v.size();
154 }
155 return sample;
156}

163inline symbol symbol_of(const encoder& e, std::uint16_t code) {
164 if (code >= kPseudo) return {static_cast<std::uint64_t>(code - kPseudo), 1};
165 return {e.symbols[code], e.lengths[code]};

173template <class ValueAt>
174bool train(std::size_t n, ValueAt&& value_at, encoder& out) {
175 using namespace enc_detail;
176 const auto sample = make_sample(n, value_at);
177 encoder table;
178 reset(table);
179 encoder best = table;
180 std::int64_t best_gain = std::numeric_limits<std::int64_t>::min();
181 std::vector<std::uint16_t> count1(kCodes);
182 std::vector<std::uint16_t> count2(kCodes * kCodes);
183 // Rows of count2 written this round, so clearing it costs what was used rather than 512 KiB.
184 std::vector<std::uint16_t> touched_rows;
185 std::vector<bool> row_touched(kCodes);
186 const auto bump = [](std::uint16_t& c) {
187 if (c != 0xFFFF) ++c;
188 };
189 const auto code_at = [&table](const std::uint8_t* p, std::size_t remaining, unsigned& length) {
190 const auto code = find(table, p, remaining, length);
191 return code == encoder::kNone ? static_cast<std::uint16_t>(kPseudo + *p) : code;
192 };
193
194 for (const unsigned frac : {8U, 38U, 68U, 98U, 108U, 128U}) {
195 // Compress the sample with the current table, counting symbols and adjacent pairs.
196 std::fill(count1.begin(), count1.end(), std::uint16_t{0});
197 for (const auto row : touched_rows) {
198 std::fill_n(count2.begin() + static_cast<std::ptrdiff_t>(row * kCodes), kCodes, std::uint16_t{0});
199 row_touched[row] = false;
200 }
201 touched_rows.clear();
202 std::int64_t gain = 0;
203 for (const auto& v : sample) {
204 const auto* data = reinterpret_cast<const std::uint8_t*>(v.data());
205 const std::size_t size = v.size();
206 if (size == 0) continue;
207 std::size_t pos = 0;
208 unsigned length = 0;
209 auto prev = code_at(data, size, length);
210 pos += length;
211 gain += std::max<std::int64_t>(0, static_cast<std::int64_t>(length) - 1 - (prev >= kPseudo ? 1 : 0));
212 while (pos < size) {
213 bump(count1[prev]);
214 if (symbol_of(table, prev).length != 1) bump(count1[kPseudo + data[pos]]);
215 const auto cur = code_at(data + pos, size - pos, length);
216 gain += std::max<std::int64_t>(0, static_cast<std::int64_t>(length) - 1 - (cur >= kPseudo ? 1 : 0));
217 if (frac < 128) { // no pairs in the last round
218 if (!row_touched[prev]) {
219 row_touched[prev] = true;
220 touched_rows.push_back(prev);
221 }
222 bump(count2[prev * kCodes + cur]);
223 if (length > 1) bump(count2[prev * kCodes + kPseudo + data[pos]]);
224 }
225 pos += length;
226 prev = cur;
227 }
228 bump(count1[prev]);
229 }
230 if (gain >= best_gain) {
231 best_gain = gain;
232 best = table;
233 }
234
235 // Candidates: every symbol seen, and every pair seen, weighted by count x length.
236 std::map<std::pair<std::uint64_t, unsigned>, std::uint64_t> candidates;
237 const auto consider = [&](symbol s, std::uint64_t count) {
238 if (count < (5U * frac) / 128U) return;
239 candidates[{s.value, s.length}] += count * s.length;
240 };
241 std::vector<std::uint16_t> live;
242 for (std::uint16_t c = 0; c < table.symbol_count; ++c) live.push_back(c);
243 for (std::uint16_t b = 0; b < 256; ++b) live.push_back(static_cast<std::uint16_t>(kPseudo + b));
244 for (const auto c1 : live) {
245 const auto n1 = count1[c1];
246 if (n1 == 0) continue;
247 const auto s1 = symbol_of(table, c1);
248 // Single bytes count 8x: they cut the escape rate (the paper's heuristic).
249 consider(s1, static_cast<std::uint64_t>(s1.length == 1 ? 8 : 1) * n1);
250 if (frac >= 128 || s1.length == 8 || !row_touched[c1]) continue;
251 for (const auto c2 : live) {
252 const auto n2 = count2[c1 * kCodes + c2];
253 if (n2 == 0) continue;
254 const auto s2 = symbol_of(table, c2);
255 const unsigned length = std::min(8U, s1.length + s2.length);
256 const auto value = (s1.value | (s2.value << (8 * s1.length))) & mask(length);
257 consider(symbol{value, length}, n2);
258 }
259 }
260 std::vector<std::pair<std::uint64_t, std::pair<std::uint64_t, unsigned>>> ranked;
261 ranked.reserve(candidates.size());
262 for (const auto& [sym, g] : candidates) ranked.emplace_back(g, sym);
263 std::sort(ranked.begin(), ranked.end(), [](const auto& a, const auto& b) {
264 return a.first != b.first ? a.first > b.first : a.second.first < b.second.first;
265 });
266 reset(table);
267 for (const auto& [g, sym] : ranked) {
268 if (table.symbol_count == kMaxSymbols) break;
269 add(table, sym.first, sym.second);
270 }
271 }
272 if (best.symbol_count == 0) return false;
273 best.short_or_byte.resize(65536);
274 for (std::uint32_t w = 0; w < 65536; ++w) {
275 const auto two = best.short_codes[w];
276 const auto one = best.byte_codes[w & 0xFF];
277 best.short_or_byte[w] = two != encoder::kNone ? static_cast<std::uint16_t>((2U << 8) | two)
278 : one != encoder::kNone ? static_cast<std::uint16_t>((1U << 8) | one)
279 : static_cast<std::uint16_t>((1U << 8) | kEscape);
280 }
281 out = std::move(best);
282 return true;
283}

292inline std::size_t compress_unchecked(const encoder& e, std::span<const std::byte> in, std::byte* dst) {
293 using namespace enc_detail;
294 const auto* data = reinterpret_cast<const std::uint8_t*>(in.data());
295 const std::size_t size = in.size();
296 auto* out = reinterpret_cast<std::uint8_t*>(dst);
297 const auto* const begin = out;
298 std::size_t pos = 0;
299 if (!e.short_or_byte.empty()) {
300 // While 8 bytes remain: one unaligned load, the long-symbol slot, then one lookup that settles
301 // the two-byte / one-byte / escape cases. The code and the literal byte are both stored and
302 // `out` advances past the literal only for an escape (it stays within 2 * size).
303 while (pos + 8 <= size) {
304 std::uint64_t word = 0;
305 std::memcpy(&word, data + pos, 8);
306 const auto& s = e.long_codes[hash3(word)];
307 if (s.length != 0 && ((word ^ s.value) << (64 - 8 * s.length)) == 0) {
308 *out++ = s.code;
309 pos += s.length;
310 continue;
311 }
312 const auto entry = e.short_or_byte[word & 0xFFFF];
313 const auto code = static_cast<std::uint8_t>(entry & 0xFF);
314 out[0] = code;
315 out[1] = static_cast<std::uint8_t>(word & 0xFF);
316 out += code == kEscape ? 2 : 1;
317 pos += entry >> 8;
318 }
319 // The last < 8 bytes, the same way from a zero-padded copy (so the loads stay in bounds), with
320 // no match allowed to run past the value.
321 if (pos < size) {
322 std::array<std::uint8_t, 16> tail{};
323 const auto rest = size - pos;
324 std::memcpy(tail.data(), data + pos, rest);
325 std::size_t at = 0;
326 while (at < rest) {
327 std::uint64_t word = 0;
328 std::memcpy(&word, tail.data() + at, 8);
329 const auto remaining = rest - at;
330 const auto& s = e.long_codes[hash3(word)];
331 if (s.length != 0 && s.length <= remaining && ((word ^ s.value) << (64 - 8 * s.length)) == 0) {
332 *out++ = s.code;
333 at += s.length;
334 continue;
335 }
336 std::uint16_t entry = 0;
337 if (remaining >= 2) {
338 entry = e.short_or_byte[word & 0xFFFF];
339 } else {
340 const auto one = e.byte_codes[word & 0xFF];
341 entry = static_cast<std::uint16_t>((1U << 8) | (one != encoder::kNone ? one : kEscape));
342 }
343 const auto code = static_cast<std::uint8_t>(entry & 0xFF);
344 out[0] = code;
345 out[1] = static_cast<std::uint8_t>(word & 0xFF);
346 out += code == kEscape ? 2 : 1;
347 at += entry >> 8;
348 }
349 }
350 return static_cast<std::size_t>(out - begin);
351 }
352 while (pos < size) { // an encoder built without train(): the plain lookups
353 unsigned length = 0;
354 const auto code = find(e, data + pos, size - pos, length);
355 if (code == encoder::kNone) {
356 *out++ = kEscape;
357 *out++ = data[pos];
358 ++pos;
359 } else {
360 *out++ = static_cast<std::uint8_t>(code);
361 pos += length;
362 }
363 }
364 return static_cast<std::size_t>(out - begin);
365}

368inline void compress(const encoder& e, std::span<const std::byte> in, std::vector<std::byte>& out) {
369 const auto at = out.size();
370 out.resize(at + max_compressed_size(in.size()));
371 out.resize(at + compress_unchecked(e, in, out.data() + at));
372}

375inline std::array<std::byte, kSymbolTableBytes> serialize(const encoder& e) {
376 std::array<std::byte, kSymbolTableBytes> out{};
377 const auto put64 = [&](std::size_t at, std::uint64_t v) {
378 if constexpr (std::endian::native == std::endian::big) v = std::byteswap(v);
379 std::memcpy(out.data() + at, &v, 8);
380 };
381 put64(0, kMagic | kEncoderSwitchBit | e.symbol_count);
382 std::size_t pos = 8;
383 for (std::uint32_t i = 0; i < e.symbol_count; ++i, pos += 8) put64(pos, e.symbols[i]);
384 for (std::uint32_t i = 0; i < e.symbol_count; ++i) out[pos++] = std::byte(e.lengths[i]);
385 return out;
386}
