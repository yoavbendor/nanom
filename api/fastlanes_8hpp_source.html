2// nanom/fastlanes.hpp — FastLanes 1024-value bit unpacking (Lance's InlineBitpacking and format 2.0
3// Bitpacked pages; the kernel Lance vendors as lance-bitpacking = spiraldb/fastlanes).
4//
5// FastLanes packs a block of exactly 1024 unsigned values of one word type T (u8, u16, u32, u64) at
6// `width` bits each, in a transposed layout so every row of the block is one SIMD-friendly loop over
7// LANES = 1024 / bits(T) words. A block packed at `width` takes packed_words_1024<T>(width) words of
8// T, i.e. 128 * width bytes, whatever T is. The value order is FastLanes' FL_ORDER: input element
9// fl_index(row, lane) sits in lane `lane` of row `row`.
10//
11// Same safety rule as columnar.hpp — CHECK AT THE BOUNDARY, RUN UNCHECKED INSIDE: unpack_1024
12// validates the width and the packed span once, then runs a loop whose width is a template constant
13// (one instantiation per width, chosen through a table) so every shift and word boundary is folded.
14// The encoder, pack_1024, is in columnar_encode.hpp; reading code does not include it.

45namespace detail {

47inline constexpr unsigned kFlOrder[8] = {0, 4, 2, 6, 1, 5, 3, 7};
48
50constexpr std::size_t fl_index(std::size_t row, std::size_t lane) {
51 return static_cast<std::size_t>(kFlOrder[row / 8U]) * 16U + (row % 8U) * 128U + lane;
52}
53
54template <word T>
55constexpr T fl_mask(unsigned width) {
56 if (width == 0U) return T(0);
57 if (width >= kBits<T>) return static_cast<T>(~T(0));
58 return static_cast<T>((T(1) << width) - T(1));
59}
60
61// Row-outer, lane-inner: for a fixed row, both fl_index(row, lane) and the packed words are
62// contiguous in `lane`, so each inner loop is a contiguous run the compiler vectorizes. Width is a
63// template constant, as in the Rust crate's macro-generated per-width functions.
64template <word T, unsigned Width>
65void unpack_w(const T* in, T* out) {
66 constexpr unsigned bits = kBits<T>;
67 constexpr std::size_t lanes = kBlock / bits;
68 T src[128]; // lanes <= 128 (the u8 case)
69 std::memcpy(src, in, lanes * sizeof(T));
70 for (unsigned row = 0; row < bits; ++row) {
71 const unsigned shift = (row * Width) % bits;
72 const unsigned curr_word = (row * Width) / bits;
73 const unsigned next_word = ((row + 1U) * Width) / bits;
74 T* out_row = out + fl_index(row, 0);
75 if (next_word > curr_word) {
76 const unsigned remaining = ((row + 1U) * Width) % bits;
77 const unsigned current_bits = Width - remaining;
78 const T low_mask = fl_mask<T>(current_bits);
79 for (std::size_t lane = 0; lane < lanes; ++lane)
80 out_row[lane] = static_cast<T>((src[lane] >> shift) & low_mask);
81 if (next_word < Width) {
82 const T* next_in = in + lanes * next_word;
83 const T high_mask = fl_mask<T>(remaining);
84 for (std::size_t lane = 0; lane < lanes; ++lane) {
85 src[lane] = next_in[lane];
86 out_row[lane] = static_cast<T>(out_row[lane] | static_cast<T>((src[lane] & high_mask) << current_bits));
87 }
88 }
89 } else {
90 constexpr T mask = fl_mask<T>(Width);
91 for (std::size_t lane = 0; lane < lanes; ++lane) out_row[lane] = static_cast<T>((src[lane] >> shift) & mask);
92 }
93 }
94}
95
96} // namespace detail

100template <word T>
101void unpack_1024_unchecked(unsigned width, const T* in, T* out) {
102 constexpr unsigned bits = kBits<T>;
103 constexpr std::size_t lanes = kBlock / bits;
104 if (width == 0U) {
105 std::memset(out, 0, kBlock * sizeof(T));
106 return;
107 }
108 if (width == bits) {
109 for (unsigned row = 0; row < bits; ++row)
110 std::memcpy(out + detail::fl_index(row, 0), in + lanes * row, lanes * sizeof(T));
111 return;
112 }
113 using fn = void (*)(const T*, T*);
114 static constexpr auto table = []<unsigned... Ws>(std::integer_sequence<unsigned, Ws...>) {
115 return std::array<fn, sizeof...(Ws)>{&detail::unpack_w<T, Ws + 1U>...};
116 }(std::make_integer_sequence<unsigned, bits - 1U>{});
117 table[width - 1U](in, out);
118}

123template <word T>
124bool unpack_1024(unsigned width, std::span<const T> packed, std::span<T, kBlock> out) {
125 if (width > kBits<T> || packed.size() < packed_words_1024<T>(width)) return false;
126 unpack_1024_unchecked<T>(width, packed.data(), out.data());
127 return true;
128}
