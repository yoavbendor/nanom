2// nanom/columnar.hpp — decode kernels for columnar page encodings (Parquet; Lance shares several).
3//
4// These are the inner loops of a column reader: they turn encoded page bytes into flat, host-order
5// value buffers. Every kernel follows one safety rule — CHECK AT THE BOUNDARY, RUN UNCHECKED INSIDE:
6// sizes and counts taken from the wire are validated once, against the input span and the caller's
7// output span, before the loop; the loop itself never reads past what was proven present and never
8// writes past the output. A kernel that cannot finish reports how far it got instead of guessing.
9//
10// unpack_bits<U> LSB-first bit-packed integers (Parquet BIT_PACKED groups, RLE-hybrid runs,
11// delta miniblocks). The bit width is a TEMPLATE parameter inside the loop:
12// one runtime switch selects a loop specialized for that width.
13// rle_bp_decoder Parquet's RLE / bit-packed hybrid (definition/repetition levels, dictionary
14// indices, RLE booleans), streaming in caller-sized batches.
15// delta_binary_packed DELTA_BINARY_PACKED (int32 / int64), also the length stream of
16// DELTA_LENGTH_BYTE_ARRAY and DELTA_BYTE_ARRAY.
17// byte_stream_split BYTE_STREAM_SPLIT (floats, doubles, any fixed width).
18// unpack_bool_bitmap PLAIN booleans: LSB-first bits, already Arrow's bitmap layout (copy/shift).

32inline std::uint64_t load_le64(const std::uint8_t* p) {
33 std::uint64_t v;
34 std::memcpy(&v, p, 8);
35 if constexpr (std::endian::native == std::endian::big) v = std::byteswap(v);
36 return v;
37}

41template <class U, unsigned W>
42inline void unpack8(const std::uint8_t* in, U* out) {
43 if constexpr (W == 0) {
44 for (int i = 0; i < 8; ++i) out[i] = 0;
45 } else if constexpr (W <= 56) {
46 constexpr std::uint64_t mask = W == 64 ? ~std::uint64_t(0) : (std::uint64_t(1) << W) - 1;
47 for (unsigned i = 0; i < 8; ++i) {
48 const unsigned bit = i * W;
49 out[i] = U((load_le64(in + bit / 8) >> (bit % 8)) & mask);
50 }
51 } else {
52 // widths 57..64 can straddle 9 bytes: assemble from two loads
53 constexpr std::uint64_t mask = W == 64 ? ~std::uint64_t(0) : (std::uint64_t(1) << W) - 1;
54 for (unsigned i = 0; i < 8; ++i) {
55 const unsigned bit = i * W, sh = bit % 8;
56 const std::uint64_t lo = load_le64(in + bit / 8);
57 const std::uint64_t hi = sh ? std::uint64_t(in[bit / 8 + 8]) << (64 - sh) : 0;
58 out[i] = U(((lo >> sh) | hi) & mask);
59 }
60 }
61}

64template <class U, unsigned W>
65inline void unpack_groups(const std::uint8_t* in, std::size_t in_len, U* out, std::size_t groups) {
66 std::size_t g = 0;
67 // fast path: enough slack for the unchecked 64-bit loads
68 const std::size_t safe = in_len >= W + 8 ? (in_len - (W + 8)) / (W ? W : 1) + 1 : 0;
69 const std::size_t fast = W == 0 ? groups : std::min(groups, safe);
70 for (; g < fast; ++g) unpack8<U, W>(in + g * W, out + g * 8);
71 // tail: copy the group into a zero-padded buffer
72 for (; g < groups; ++g) {
73 std::uint8_t buf[64 + 16]{};
74 const std::size_t at = g * W;
75 const std::size_t have = at < in_len ? std::min<std::size_t>(W, in_len - at) : 0;
76 if (have) std::memcpy(buf, in + at, have);
77 unpack8<U, W>(buf, out + g * 8);
78 }
79}

96template <class U>
97 requires(std::same_as<U, std::uint32_t> || std::same_as<U, std::uint64_t>)
98inline bool unpack_bits(std::span<const std::byte> in, unsigned width, std::span<U> out, std::size_t n) {
99 if (width > 8 * sizeof(U) || out.size() < n) return false;
100 const std::uint64_t need_bits = std::uint64_t(n) * width;
101 if (in.size() < (need_bits + 7) / 8) return false;
102 const auto* src = reinterpret_cast<const std::uint8_t*>(in.data());
103 const std::size_t full = n / 8;
104 detail::unpack_table<U>[width](src, in.size(), out.data(), full);
105 if (const std::size_t rem = n % 8) {
106 U tmp[8];
107 const std::size_t at = full * width;
108 detail::unpack_table<U>[width](src + at, in.size() - at, tmp, 1);
109 for (std::size_t i = 0; i < rem; ++i) out[full * 8 + i] = tmp[i];
110 }
111 return true;
112}

126class rle_bp_decoder {
127 public:
128 rle_bp_decoder() = default;
129 rle_bp_decoder(std::span<const std::byte> data, unsigned width)
130 : p_(data.data()), e_(data.data() + data.size()), width_(width), ok_(width <= 32) {}
131
132 bool ok() const { return ok_; }
133 unsigned width() const { return width_; }
134
135 std::size_t get(std::uint32_t* out, std::size_t n) {
136 std::size_t done = 0;
137 while (done < n && ok_) {
138 if (rle_left_) {
139 const std::size_t k = std::min<std::size_t>(rle_left_, n - done);
140 std::fill_n(out + done, k, rle_value_);
141 rle_left_ -= k;
142 done += k;
143 } else if (bp_left_) {
144 done += take_bitpacked(out + done, n - done);
145 } else if (!next_run()) {
146 break;
147 }
148 }
149 return done;
150 }
151
155 struct run {
156 bool packed = false;
157 std::uint32_t value = 0;
158 std::size_t count = 0;
159 std::span<const std::byte> bits;
160 std::size_t bit_offset = 0;
161 };
164 bool next_run(run& r, std::size_t max_values) {
165 while (ok_ && max_values) {
166 if (rle_left_) {
167 r.packed = false;
168 r.value = rle_value_;
169 r.count = std::min(rle_left_, max_values);
170 rle_left_ -= r.count;
171 return true;
172 }
173 if (bp_left_) {
174 r.packed = true;
175 r.count = std::min(bp_left_, max_values);
176 const std::size_t bit = bp_pos_ * width_;
177 r.bits = std::span<const std::byte>(bp_ + bit / 8, bp_bytes_ - bit / 8);
178 r.bit_offset = bit % 8;
179 bp_pos_ += r.count;
180 bp_left_ -= r.count;
181 return true;
182 }
183 if (!next_run()) return false;
184 }
185 return false;
186 }
187
189 std::size_t skip(std::size_t n) {
190 std::uint32_t buf[256];
191 std::size_t done = 0;
192 while (done < n) {
193 const std::size_t k = get(buf, std::min<std::size_t>(256, n - done));
194 if (!k) break;
195 done += k;
196 }
197 return done;
198 }
199
200 private:
201 bool next_run() {
202 if (p_ >= e_) return false;
203 std::uint64_t h = 0;
204 {
205 const input in = from(std::span<const std::byte>(p_, std::size_t(e_ - p_)));
206 auto r = varint_u64(in);
207 if (!r) { ok_ = false; return false; }
208 h = r->value;
209 p_ = r->rest.first;
210 }
211 if (h & 1) {
212 const std::uint64_t groups = h >> 1;
213 // values actually present: whole bytes available limit the run
214 const std::size_t bytes_avail = std::size_t(e_ - p_);
215 // groups comes from the wire: clamp before multiplying (each group is >= 1 byte at width > 0)
216 const std::uint64_t want_bytes = groups > bytes_avail ? std::uint64_t(bytes_avail) * width_ : groups * width_;
217 const std::size_t run_bytes = std::size_t(std::min<std::uint64_t>(want_bytes, bytes_avail));
218 bp_left_ = width_ ? std::size_t(std::min<std::uint64_t>(groups * 8, std::uint64_t(run_bytes) * 8 / width_))
219 : std::size_t(std::min<std::uint64_t>(groups * 8, std::uint64_t(1) << 40));
220 bp_ = p_;
221 bp_bytes_ = run_bytes;
222 bp_pos_ = 0;
223 p_ += run_bytes;
224 if (groups == 0) return true; // empty run: legal, move on
225 } else {
226 const std::size_t vbytes = (width_ + 7) / 8;
227 if (std::size_t(e_ - p_) < vbytes) { ok_ = false; return false; }
228 std::uint32_t v = 0;
229 for (std::size_t i = 0; i < vbytes; ++i) v |= std::uint32_t(std::uint8_t(p_[i])) << (8 * i);
230 if (width_ < 32 && (v >> width_) != 0) { ok_ = false; return false; } // value wider than width
231 p_ += vbytes;
232 rle_value_ = v;
233 rle_left_ = std::size_t(std::min<std::uint64_t>(h >> 1, std::uint64_t(1) << 40));
234 }
235 return true;
236 }
237
238 std::size_t take_bitpacked(std::uint32_t* out, std::size_t n) {
239 // bp_pos_ counts values consumed from this run; decode whole groups through a small buffer
240 // when not group-aligned, directly into out otherwise
241 std::size_t done = 0;
242 while (done < n && bp_left_) {
243 const std::size_t in_group = bp_pos_ % 8;
244 const std::size_t group_byte = (bp_pos_ / 8) * width_;
245 const auto in = std::span<const std::byte>(bp_ + group_byte, bp_bytes_ - group_byte);
246 if (in_group == 0 && n - done >= 8 && bp_left_ >= 8) {
247 const std::size_t groups = std::min(n - done, bp_left_) / 8;
248 unpack_bits<std::uint32_t>(in, width_, std::span<std::uint32_t>(out + done, groups * 8), groups * 8);
249 done += groups * 8;
250 bp_pos_ += groups * 8;
251 bp_left_ -= groups * 8;
252 } else {
253 std::uint32_t g[8];
254 const std::size_t avail_in_group = std::min<std::size_t>(8, bp_left_ + in_group);
255 detail::unpack_table<std::uint32_t>[width_](reinterpret_cast<const std::uint8_t*>(in.data()),
256 in.size(), g, 1);
257 const std::size_t k = std::min(avail_in_group - in_group, n - done);
258 for (std::size_t i = 0; i < k; ++i) out[done + i] = g[in_group + i];
259 done += k;
260 bp_pos_ += k;
261 bp_left_ -= k;
262 }
263 }
264 return done;
265 }
266
267 const std::byte* p_ = nullptr;
268 const std::byte* e_ = nullptr;
269 unsigned width_ = 0;
270 bool ok_ = false;
271 std::uint32_t rle_value_ = 0;
272 std::size_t rle_left_ = 0;
273 const std::byte* bp_ = nullptr;
274 std::size_t bp_bytes_ = 0, bp_pos_ = 0, bp_left_ = 0;
275};

286template <class T>
287 requires(std::same_as<T, std::int32_t> || std::same_as<T, std::int64_t>)
288inline std::optional<std::size_t> delta_binary_packed(std::span<const std::byte> in, std::span<T> out,
289 std::size_t n) {
290 using U = std::make_unsigned_t<T>;
291 if (out.size() < n) return std::nullopt;
292 input cur = from(in);
293 auto rd = [&](std::uint64_t& v) {
294 auto r = varint_u64(cur);
295 if (!r) return false;
296 v = r->value;
297 cur = r->rest;
298 return true;
299 };
300 std::uint64_t block_size = 0, minis = 0, total = 0, first_zz = 0;
301 if (!rd(block_size) || !rd(minis) || !rd(total) || !rd(first_zz)) return std::nullopt;
302 if (block_size == 0 || block_size % 128 != 0 || minis == 0 || block_size % minis != 0 ||
303 (block_size / minis) % 32 != 0 || block_size > (1u << 20))
304 return std::nullopt;
305 if (total < n) return std::nullopt;
306 const std::size_t per_mini = std::size_t(block_size / minis);
307 if (n == 0) return std::size_t(cur.first - in.data());
308 U last = U(zigzag_decode(first_zz));
309 out[0] = std::bit_cast<T>(last);
310 std::size_t produced = 1;
311 std::uint64_t unpacked[1024]; // miniblocks are decoded in chunks of at most 1024 values
312 while (produced < n) {
313 std::uint64_t min_zz = 0;
314 if (!rd(min_zz)) return std::nullopt;
315 const U min_delta = U(zigzag_decode(min_zz));
316 if (cur.size() < minis) return std::nullopt;
317 const std::byte* widths = cur.first;
318 cur = cur.advance(std::size_t(minis));
319 for (std::size_t m = 0; m < minis && produced < n; ++m) {
320 const unsigned w = std::uint8_t(widths[m]);
321 if (w > 8 * sizeof(T)) return std::nullopt;
322 const std::size_t mini_bytes = per_mini * w / 8;
323 const std::size_t need_vals = std::min(per_mini, n - produced);
324 const std::size_t need_bytes = (need_vals * w + 7) / 8;
325 if (cur.size() < need_bytes) return std::nullopt;
326 // decode in chunks of up to 1024 values (per_mini may exceed that for huge blocks)
327 std::size_t done = 0;
328 while (done < need_vals) {
329 const std::size_t k = std::min<std::size_t>(1024, need_vals - done);
330 const std::size_t off_bytes = done * w / 8; // done is a multiple of 1024 -> byte aligned
331 if (!unpack_bits<std::uint64_t>(std::span<const std::byte>(cur.first + off_bytes, cur.size() - off_bytes), w,
332 std::span<std::uint64_t>(unpacked, k), k))
333 return std::nullopt;
334 for (std::size_t i = 0; i < k; ++i) {
335 last = U(last + min_delta + U(unpacked[i]));
336 out[produced++] = std::bit_cast<T>(last);
337 }
338 done += k;
339 }
340 // a miniblock that held values is padded to full size on the wire (the last may be cut short)
341 cur = cur.advance(std::min(mini_bytes, cur.size()));
342 }
343 }
344 return std::size_t(cur.first - in.data());
345}

352inline bool byte_stream_split(std::span<const std::byte> in, std::size_t width, std::size_t n,
353 std::span<std::byte> out) {
354 if (width == 0 || n > std::numeric_limits<std::size_t>::max() / width) return false;
355 const std::size_t total = n * width;
356 if (in.size() < total || out.size() < total) return false;
357 const std::byte* src = in.data();
358 std::byte* dst = out.data();
359 if (width == 4) { // the float case, unrolled
360 for (std::size_t i = 0; i < n; ++i) {
361 dst[4 * i + 0] = src[i];
362 dst[4 * i + 1] = src[n + i];
363 dst[4 * i + 2] = src[2 * n + i];
364 dst[4 * i + 3] = src[3 * n + i];
365 }
366 } else if (width == 8) {
367 for (std::size_t i = 0; i < n; ++i)
368 for (std::size_t b = 0; b < 8; ++b) dst[8 * i + b] = src[b * n + i];
369 } else {
370 for (std::size_t b = 0; b < width; ++b)
371 for (std::size_t i = 0; i < n; ++i) dst[width * i + b] = src[b * n + i];
372 }
373 return true;
374}

379inline bool copy_bits(std::span<const std::byte> in, std::size_t n, std::span<std::byte> out,
380 std::size_t dst_bit) {
381 if (in.size() < (n + 7) / 8) return false;
382 if (dst_bit > std::numeric_limits<std::size_t>::max() - n || out.size() < (dst_bit + n + 7) / 8)
383 return false;
384 if (n == 0) return true;
385 const auto* s = reinterpret_cast<const std::uint8_t*>(in.data());
386 auto* d = reinterpret_cast<std::uint8_t*>(out.data());
387 if (dst_bit % 8 == 0) {
388 const std::size_t full = n / 8;
389 std::memcpy(d + dst_bit / 8, s, full);
390 if (const std::size_t rem = n % 8) {
391 const std::uint8_t mask = std::uint8_t((1u << rem) - 1);
392 std::uint8_t& t = d[dst_bit / 8 + full];
393 t = std::uint8_t((t & ~mask) | (s[full] & mask));
394 }
395 return true;
396 }
397 // unaligned destination: every output byte takes the high bits of one source byte and the low
398 // bits of the next. Fill the partial first byte, then whole output words, then the tail.
399 const unsigned sh = unsigned(dst_bit % 8); // destination bit offset in its byte
400 std::uint8_t* o = d + dst_bit / 8;
401 {
402 const std::size_t head = std::min<std::size_t>(n, 8 - sh);
403 const std::uint8_t mask = std::uint8_t(((1u << head) - 1) << sh);
404 *o = std::uint8_t((*o & ~mask) | ((s[0] << sh) & mask));
405 if (n == head) return true;
406 ++o;
407 }
408 // the rest is byte-aligned in the output: output byte q holds source bits [8q + k, 8q + k + 8)
409 const unsigned k = 8 - sh; // 1..7 source bits went into the first byte
410 std::size_t left = n - k, p = 0;
411 const std::size_t in_bytes = (n + 7) / 8;
412 while (left >= 64 && p + 9 <= in_bytes) {
413 std::uint64_t w;
414 std::memcpy(&w, s + p, 8);
415 w = (w >> k) | (std::uint64_t(s[p + 8]) << (64 - k));
416 std::memcpy(o, &w, 8);
417 o += 8;
418 p += 8;
419 left -= 64;
420 }
421 while (left >= 8) { // whole bytes: s[p + 1] exists because k + 8 more bits remain in the source
422 *o++ = std::uint8_t((s[p] >> k) | (s[p + 1] << (8 - k)));
423 ++p;
424 left -= 8;
425 }
426 if (left) {
427 const std::size_t first = p * 8 + k; // source bit of the tail's first bit
428 unsigned v = s[first / 8] >> (first % 8);
429 if ((first % 8) + left > 8) v |= unsigned(s[first / 8 + 1]) << (8 - first % 8);
430 const std::uint8_t mask = std::uint8_t((1u << left) - 1);
431 *o = std::uint8_t((*o & ~mask) | (v & mask));
432 }
433 return true;
434}
