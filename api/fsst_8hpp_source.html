2// nanom/fsst.hpp — FSST string decompression (Fast Static Symbol Table, Boncz / Neumann / Leis 2020),
3// with the symbol table serialized as Lance writes it (rust/compression/fsst in lancedb/lance).
4//
5// FSST compresses a string column with one table of up to 255 symbols of 1..8 bytes. A compressed
6// value is a stream of codes: 255 escapes the next byte as a literal, any other code emits its
7// symbol. Lance stores the table as a fixed kSymbolTableBytes (2312) bytes:
8// [u64 header][n * u64 symbol][n * u8 length], zero-padded,
9// header = "FSST" << 32 | encoder_switch << 24 | suffix_lim << 16 | terminator << 8 | n.
10// encoder_switch 0 means the writer declined to compress (Lance skips inputs under 32 KiB): the
11// "compressed" values are the originals and decode copies them through.
12//
13// The table, the codes and their lengths come from the file, so everything is checked before use:
14// the table's size and magic, every declared symbol length (1..8), every code against the symbol
15// count, and an escape with no byte after it. An undeclared code is an error, not an empty symbol.
16// Integers are little-endian (Lance writes them native-endian, which is little-endian in practice).
17//
18// The encoder (training and compression) is in fsst_encode.hpp; reading code does not include it.

59inline std::uint64_t load_le64(const std::byte* p) {
60 std::uint64_t v;
61 std::memcpy(&v, p, 8);
62 if constexpr (std::endian::native == std::endian::big) v = std::byteswap(v);
63 return v;
64}

69inline status parse_symbol_table(std::span<const std::byte> bytes, symbol_table& out) {
70 if (bytes.size() != kSymbolTableBytes) return codec::detail::fail("fsst: symbol table size is not 2312 bytes", bytes.size());
71 const std::uint64_t header = detail::load_le64(bytes.data());
72 if ((header & kMagic) != kMagic) return codec::detail::fail("fsst: symbol table has the wrong magic", 0);
73 symbol_table t;
74 t.passthrough = (header & kEncoderSwitchBit) == 0;
75 t.symbol_count = static_cast<std::uint32_t>(header & 0xFF);
76 // Symbols first, then one length byte per symbol right after them (not after all 256 slots).
77 std::size_t pos = 8;
78 for (std::uint32_t i = 0; i < t.symbol_count; ++i, pos += kMaxSymbolLength)
79 std::memcpy(t.symbols[i].data(), bytes.data() + pos, kMaxSymbolLength);
80 for (std::uint32_t i = 0; i < t.symbol_count; ++i, ++pos) {
81 const auto length = std::uint8_t(bytes[pos]);
82 if (length < 1 || length > kMaxSymbolLength) return codec::detail::fail("fsst: symbol length is not 1..8", pos);
83 t.lengths[i] = length;
84 }
85 out = t;
86 return kSymbolTableBytes;
87}

93inline bool decode_unchecked(const symbol_table& table, std::span<const std::byte> in, std::byte*& dst,
94 codec_error* error = nullptr) {
95 const auto* data = reinterpret_cast<const std::uint8_t*>(in.data());
96 const std::size_t size = in.size();
97 if (table.passthrough) {
98 if (size) std::memcpy(dst, data, size);
99 dst += size;
100 return true;
101 }
102 std::byte* out = dst;
103 for (std::size_t i = 0; i < size;) {
104 const std::uint8_t code = data[i];
105 if (code == kEscape) {
106 if (i + 1 >= size) {
107 if (error) *error = codec_error{"fsst: escape at the end of a value has no byte", i};
108 return false;
109 }
110 *out++ = std::byte(data[i + 1]);
111 i += 2;
112 continue;
113 }
114 if (code >= table.symbol_count) {
115 if (error) *error = codec_error{"fsst: code is not in the symbol table", i};
116 return false;
117 }
118 std::memcpy(out, table.symbols[code].data(), kMaxSymbolLength);
119 out += table.lengths[code];
120 ++i;
121 }
122 dst = out;
123 return true;
124}

128inline status decode(const symbol_table& table, std::span<const std::byte> in, std::span<std::byte> out) {
129 if (out.size() < max_decoded_size(in.size())) return codec::detail::fail("fsst: output is smaller than max_decoded_size", 0);
130 std::byte* dst = out.data();
131 codec_error error;
132 if (!decode_unchecked(table, in, dst, &error)) return unexpected<codec_error>(error);
133 return std::size_t(dst - out.data());
134}
